### Grid.Load (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&Update = !'<i class="fa fa-pen"></i>'
	If (&IsAuthorized_Update)
		&Update.Link = Calidad.Reclamo.Link(TrnMode.Update, ReclamoId)
	Endif
	&Delete = !'<i class="fa fa-times"></i>'
	If (&IsAuthorized_Delete)
		&Delete.Link = Calidad.Reclamo.Link(TrnMode.Delete, ReclamoId)
	Endif
	&ActGroup.Clear()
	&ActGroup.AddItem(0, !";fa fa-cogs")
	If (&IsAuthorized_AddDetalle)
		&ActGroup.AddItem(1, format(!'%1;%2', "Agregar Carrete", !"fas fa-cog"))
	Endif
	If (&IsAuthorized_Imprimir)
		&ActGroup.AddItem(2, format(!'%1;%2', "Formato Reporte", !"fas fa-print"))
	Endif
	If (&IsAuthorized_FormatoTabular)
		&ActGroup.AddItem(3, format(!'%1;%2', "Formato Tabular", !"fas fa-table"))
	Endif
	If &ActGroup.Count = 1
		&ActGroup.Class = !'Invisible'
	Else
		&ActGroup.Class = !'ConvertToDDO'
	EndIf
	
	&DetailWebComponent = !'<i class="fas fa-info-circle"></i>'
	
	Do Case
		Case ReclamoEstatus = EstadoReclamo.Abierto
			&GridBadge = "Abierto"
			&GridBadge.TooltipText = "Abierto"
			&GridBadge.Class = !' AttributeTagWarning BootstrapTooltipLeft'
		Case ReclamoEstatus = EstadoReclamo.Cerrado
			&GridBadge = "Cerrado"
			&GridBadge.TooltipText = "Cerrado"
			&GridBadge.Class = !' AttributeTagSuccess BootstrapTooltipLeft'
		Otherwise
			&GridBadge = ''
			&GridBadge.TooltipText = ''
			&GridBadge.Class = !''
	EndCase
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

### ddo_ManageFilters.OnOptionClicked (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	Do Case
		Case ddo_ManageFilters.ActiveEventKey = !'<#Clean#>'
			Do 'CleanFilters'
			Grid.FirstPage()
			Refresh
		Case ddo_ManageFilters.ActiveEventKey = !'<#Save#>'
			Do 'SaveGridState'
			SaveFilterAs.Popup(!'Calidad.reclamoswwFilters', &PgmName + !"GridState")
			&ManageFiltersExecutionStep = 2
			Refresh
		Case ddo_ManageFilters.ActiveEventKey = !'<#Manage#>'
			ManageFilters.Popup(!'Calidad.reclamoswwFilters')
			&ManageFiltersExecutionStep = 2
			Refresh
		Otherwise
			&ManageFiltersXml = WWPBaseObjects.GetFilterByName(!'Calidad.reclamoswwFilters', ddo_ManageFilters.ActiveEventKey)
			If &ManageFiltersXml.IsEmpty()
				msg('WWP_FilterNotExist')
			Else
				Do 'CleanFilters'
				WWPBaseObjects.SaveGridState.Call(&PgmName + !"GridState",  &ManageFiltersXml)
				&GridState.FromXml(&ManageFiltersXml)
				&OrderedBy = &GridState.OrderedBy
				&OrderedDsc = &GridState.OrderedDsc
				Do 'SetDDOSortedStatus' 
				Do 'LoadRegFiltersState'
				Grid.FirstPage()
				Refresh
			EndIf
	EndCase
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

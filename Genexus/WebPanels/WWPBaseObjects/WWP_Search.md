### FSResults.Load (Event)

```genexus
DisplayType1_Cell.Visible = False
	DisplayType2_Cell.Visible = False
	DisplayType3_Cell.Visible = False
	DisplayType4_Cell.Visible = False
	DisplayType5_Cell.Visible = False
	Do Case
		Case &WWP_SearchResults.CurrentItem.Result.CurrentItem.DisplayType = !"Title"
			DisplayType1_Cell.Visible = True
			&DisplayType1_Title = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description
		Case &WWP_SearchResults.CurrentItem.Result.CurrentItem.DisplayType = !"Title and icon image"
			DisplayType2_Cell.Visible = True
			&DisplayType2_Title = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description
			DisplayType2_Icon.Caption = format(!"<i class='SearchResultIcon %1'></i>", &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2)
		Case &WWP_SearchResults.CurrentItem.Result.CurrentItem.DisplayType = !"Title and subtitle"
			DisplayType3_Cell.Visible = True
			&DisplayType3_Title = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description
			&DisplayType3_Subtitle = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2
		Case &WWP_SearchResults.CurrentItem.Result.CurrentItem.DisplayType = !"Title and image"
			DisplayType4_Cell.Visible = True
			&DisplayType4_Title = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description
			If &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2.IsEmpty()
				&DisplayType4_Image.SetEmpty()
			Else
				&DisplayType4_Image.FromURL(&WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2)
			EndIf
			&DisplayType4_Image.Visible = (not &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2.IsEmpty())
		Case &WWP_SearchResults.CurrentItem.Result.CurrentItem.DisplayType = !"Title, subtitle and icon image"
			DisplayType5_Cell.Visible = True
			&DisplayType5_Title = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description
			&DisplayType5_Subtitle = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description2
			DisplayType5_Icon.Caption = format(!"<i class='SearchResultIconWS %1'></i>", &WWP_SearchResults.CurrentItem.Result.CurrentItem.Description3)
	EndCase
	&Url = &WWP_SearchResults.CurrentItem.Result.CurrentItem.Url
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	//this code should be inside the For Each used to load the grid
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

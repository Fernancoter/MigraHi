### Grid.Load (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	//this code should be inside the For Each used to load the data
	
	//Sample conditions to be added manually:
	//Where <Attribute> = &DisplayInheritRoles when not &DisplayInheritRoles.IsEmpty()
	
	&Update = !'<i class="fa fa-pen"></i>'
	
	&Delete = !'<i class="fa fa-times"></i>'
	
	&BtnMainRole = "WWP_GAM_SetAsMain"
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
	&GAMUser.Load(&UserId)
  	&GridPageSize = Grid.Rows
	If &DisplayInheritRoles = False
		&GAMRoles = &GAMUser.GetRoles(&Errors)
		If &GAMRoles.Count = 0
			&GridPageCount = 0
		Else
			&GridPageCount = (&GAMRoles.Count/&GridPageSize) + iif(Mod(&GAMRoles.Count, &GridPageSize) > 0, 1, 0)
		EndIf
		&GridRecordCount = &GAMRoles.Count
		For &GAMRole in &GAMRoles
			&Name = &GAMRole.Name
			&Delete.Visible = True
			If &GAMRole.Id = &GAMUser.DefaultRoleId
				&BtnMainRole = ""
				&Name = &Name + "WWP_GAM_Main"
			Else
				&BtnMainRole = "WWP_GAM_SetAsMain"
			EndIf
			&GUID = &Role.GUID
			&Id	  = &GAMRole.Id
			
			Grid.Load()
		EndFor
	Else
		&GAMRolesDirect = &GAMUser.GetRoles(&Errors)
		&GAMRoles = &GAMUser.GetAllRoles(&Errors)
		If &GAMRolesDirect.Count > 0
			&GridPageCount = (&GAMRoles.Count/&GridPageSize) + iif(Mod(&GAMRoles.Count, &GridPageSize) > 0, 1, 0)
			&GridRecordCount = &GAMRoles.Count
			For &GAMRole in &GAMRoles
				&isDirectRole = False
				For &GAMRoleAux IN &GAMRolesDirect
					If &GAMRole.Id = &GAMRoleAux.Id
						&isDirectRole = True
						Exit
					EndIf
				EndFor
				If &isDirectRole
					&Delete.Visible = false
					If &GAMRole.Id = &GAMUser.DefaultRoleId
						&BtnMainRole = "WWP_GAM_Main"
					Else
						&BtnMainRole = "WWP_GAM_SetAsMain"
					EndIf
				Else
					&Delete.Visible = False
					&BtnMainRole.Visible	= False
				EndIf			
				&GUID = &Role.GUID
				&Id	  = &GAMRole.Id
				&Name = &GAMRole.Name
				Grid.Load()
			EndFor
		EndIf
	EndIf
```

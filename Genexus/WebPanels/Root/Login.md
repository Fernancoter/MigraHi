### Start (Event)

```genexus
LinkAPK.Link = link('HICONE.apk')
	LayoutMainTable.Class = ThemeClass:MainContainerNoBackground
	RegisterUser.Link = GAMRegisterUser.Link()
	CurrentRepository.Visible	= False
	TableButtons.Visible		= False
	
	//Valid current connection
	&isConnectionOK = GeneXusSecurity.GAM.CheckConnection()
	If GeneXusSecurity.GAM.isMultitenant()
		Do 'isMultitenantInstallation'
	Else
		If not &isConnectionOK
			If GeneXusSecurity.GAM.GetDefaultRepository(&RepositoryGUID)
				&isConnectionOK = GeneXusSecurity.GAM.SetConnectionByRepositoryGUID(&RepositoryGUID, &Errors)
			Else
				&ConnectionInfoCollection = GeneXusSecurity.GAM.GetConnections()
				If &ConnectionInfoCollection.Count > 0
					//The first connection found is established by default
					&isConnectionOK = GeneXusSecurity.GAM.SetConnection(&ConnectionInfoCollection.Item(1).Name, &Errors)
				EndIf
			Endif
		Endif
	Endif
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	Do 'AttributesSecurityCode'
	&URL.Visible = False
	&TypeAuthType.Visible = False
	&NameAuthType.Visible = False
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

### Enter (Event)

```genexus
//Set Remember User type
	Do Case
	Case &KeepMeLoggedIn
		&AdditionalParameter.RememberUserType = GAMRememberUserTypes.Authentication
	Case &RememberMe
		&AdditionalParameter.RememberUserType = GAMRememberUserTypes.Login
	Otherwise
		&AdditionalParameter.RememberUserType = GAMRememberUserTypes.None
	EndCase
	////////////////////////////////////////////////////////////
	//Send Custom Properties (optional) ////////////////////////
	//&CustomGAMProperty = new()
	//&CustomGAMProperty.Id		= "Company"
	//&CustomGAMProperty.Value	= "UY"
	//&AdditionalParameter.Properties.Add(&CustomGAMProperty)
	//These properties are saved in the session when the user authenticates
	////////////////////////////////////////////////////////////
	&AdditionalParameter.AuthenticationTypeName = &LogOnTo
	&AdditionalParameter.OTPStep				= 1
	&LoginOK = GeneXusSecurity.GAMRepository.Login(&UserName, &UserPassword, &AdditionalParameter, &Errors )
	If &LoginOK
		
		&GamUser = GamUser.Get()
		&GamUserAttribute = &GamUser.GetAttribute("CompanyId", &Errors)
		&WWPContext.CompanyId = &GamUserAttribute.Value.ToNumeric()
		
		&GamUserAttribute = &GamUser.GetAttribute("OperadorId", &Errors)
		&WWPContext.OperadorId = &GamUserAttribute.Value.ToNumeric()
		
		LoadWWPContext(&WWPContext)
		&WWPContext.CompanyId = &GamUserAttribute.Value.ToNumeric()
		SetWWPContext(&WWPContext)
		
		
		
		&URL = GAMRepository.GetLastErrorsURL()
		If &URL.IsEmpty()
			home()
		Else
			Link(&URL)
		Endif
	Else
		If &Errors.Count > 0 
			Do Case
			Case (&Errors.Item(1).Code = GAMErrorMessages.UserPasswordExpired  OR  &Errors.Item(1).Code = GAMErrorMessages.UserPasswordMustBeChanged)
				//Webpanel to Change Password
				GAMChangePassword(&IDP_State)
			Case &Errors.Item(1).Code = GAMErrorMessages.UserMissingRequiredData
				//Webpanel to Complete User data
				GAMUpdateRegisterUser(&IDP_State)
			Case &Errors.Item(1).Code = GAMErrorMessages.UserAccessCodeSent  OR  &Errors.Item(1).Code = GAMErrorMessages.UserMustValidateSecondFactorToFinishAuthentication
				//Webpanel to validate access code
				GAMOTPAuthentication(&IDP_State)
			Otherwise
				&UserPassword.SetEmpty()
				Do 'DisplayMessages'
			EndCase
		EndIf
	Endif
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

### Refresh (Event)

```genexus
TableLoginError.Visible = false
	&isRedirect = False
  
	//Gets last error in the GAM 
	&ErrorsLogin = GAMRepository.GetLastErrors()
	If &ErrorsLogin.Count > 0
		Do Case
		Case &ErrorsLogin.Item(1).Code = GAMErrorMessages.ConnectionNotSpecified  OR  &ErrorsLogin.Item(1).Code = GAMErrorMessages.UserMustBeAuthenticated
			//DO NOT display these messages
		Otherwise
			&UserPassword.SetEmpty()
			&Errors = &ErrorsLogin
			Do 'DisplayMessages'
		EndCase
	Endif
	If not &isRedirect
		//Gets last error in GAM
		&SessionValid = GAMSession.IsValid(&Session, &Errors)
		If &SessionValid and not &Session.IsAnonymous
			&URL = GAMRepository.GetLastErrorsURL()
			If &URL.IsEmpty()
				&GAMApplicationCollection = GAMRepository.GetApplications(&GAMApplicationFilter,&Errors)
				For &GAMApplication in &GAMApplicationCollection
					If &GAMApplication.GUID <> GAMInternalGUIDs.AppGAMUserBackend
						Exit
					Endif
				EndFor
				&GAMApplication =	&GAMApplication.Get()
				If &GAMApplication.HomeObject <> ''
					GAMRepository.ApplicationGoHome(&GAMApplication.GUID)
				Else
					Home.Link()
				Endif
			Else
				Link(&URL)
			Endif
		Else
			//List Log on to
			&LogOnTo.Clear()
			&AuthenticationTypes = GAMRepository.GetEnabledAuthenticationTypes(&Language, &Errors)
			For &AuthenticationType in &AuthenticationTypes
				If &AuthenticationType.NeedUserName //Authentication types you must enter the username in this web panel
					&LogOnTo.AddItem(&AuthenticationType.Name, &AuthenticationType.Description)
				Else
					GridAuthTypes.Visible = True
				EndIf
			EndFor
			If &LogOnTo.Count 	<= 1
				&LogOnTo.Visible = False
			Else
				&LogOnTo = &AuthenticationTypes.Item(1).Name
			Endif
			//Initialize variables if the user chooses to keep that information 
			&isOK = GAMRepository.GetRememberLogin(&LogOnTo, &AuxUserName, &UserRememberMe, &Errors)
			If not &AuxUserName.IsEmpty()
				&UserName = &AuxUserName
			Endif
			If &UserRememberMe = GAMRememberUserTypes.Login
				&RememberMe = True
			Endif
			//Select Default Authentication Type, Display Remember User if it is enabled on Repository
			&Repository = GAMRepository.Get()
			If &LogOnTo.Count > 1
				&LogOnTo = &Repository.DefaultAuthenticationTypeName
			Endif
			Do 'DisplayCheckBox'
			For &AuthenticationType in &AuthenticationTypes
				If &AuthenticationType.Name = &LogOnTo
					Do "ValidLOgOnToOTP"
					Exit
				Endif
			EndFor
		EndIf
	Endif
```

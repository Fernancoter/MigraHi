### Enter (Event)

```genexus
Do 'CheckRequiredFields'
	if &CheckRequiredFieldsResult
		&Repository = GAMRepository.Get()
		If &Password = &PasswordConf
			&User.Name			= &Name
			&User.EMail			= &EMail
			&User.FirstName		= &FirstName
			&User.LastName		= &LastName		
			&User.Birthday		= &Birthday
			&User.Gender.FromString(&Gender)
			&User.Password		= &Password
			&User.Save()
			If &User.Success()
        
				Commit
				If &Repository.UserActivationMethod = GeneXusSecurityCommon.GAMUserActivationMethod.Automatic
					&AdditionalParameter.RememberUserType = &UserRememberMe
					&LoginOK = GAMRepository.Login(&Name, &Password, &AdditionalParameter, &Errors )
					If &LoginOK
						GAMRepository.ApplicationGoHome(GAMInternalGUIDs.AppGAMUserBackend)
					Else
						Do 'DisplayMessages'
					EndIf
				Else
					&LinkURL = GAMRepository.ApplicationGetAccountActivationURL("")
					If &LinkURL.IsEmpty()
						&GAMApplication = GAMApplication.Get()
						&LinkURL = &GAMApplication.Environment.URL.Trim() + GAMActivateUserAccount.Link("")
						&LinkURL+= !"?%1"
					Endif
					GAMCheckUserActivationMethod(&User.GUID, &LinkURL, &Messages)
					For &Message in &Messages
						Msg(&Message.Description)
					EndFor				
				Endif
			Else
				&Errors = &User.GetErrors()
				Do 'DisplayMessages'
			Endif
		Else
			Msg("WWP_GAM_PasswordNotMatch")
		Endif
	Endif
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

```genexus
	Do Case
		Case DDO_Grid.ActiveEventKey = !'<#OrderASC#>' OR DDO_Grid.ActiveEventKey = !'<#OrderDSC#>'
			&OrderedBy.FromString(DDO_Grid.SelectedValue_get)
			&OrderedDsc = iif(DDO_Grid.ActiveEventKey = !'<#OrderDSC#>', true, false)

					&TFExtrusionExtrusoraNombre_Sel = DDO_Grid.SelectedValue_get
```

### Grid.Load (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&Update = !'<i class="fa fa-pen"></i>'
	If (&IsAuthorized_Update)
		&Update.Link = DB.Extrusion.Link(TrnMode.Update, ExtrusionId)
	Endif
	&Delete = !'<i class="fa fa-times"></i>'
	If (&IsAuthorized_Delete)
		&Delete.Link = DB.Extrusion.Link(TrnMode.Delete, ExtrusionId)
	Endif
	&DetailWebComponent = !'<i class="fas fa-info-circle"></i>'
	If (ExtrusionEstado = EstadoExtrusion.Terminada)
		&DetailWebComponent.Class = !'Attribute'
	Else
		&DetailWebComponent.Class = !'Invisible'
	Endif
	If (&IsAuthorized_ExtrusionExtrusoraNombre)
		ExtrusionExtrusoraNombre.Link = DB.ExtrusionView.Link(ExtrusionId, '')
	Endif
	Do Case
		Case ExtrusionEstado = EstadoExtrusion.Programada
			&GridBadge = "Programada"
			&GridBadge.TooltipText = "Programada"
			&GridBadge.Class = !' AttributeTagInfo BootstrapTooltipRight'
		Case ExtrusionEstado = EstadoExtrusion.Terminada
			&GridBadge = "Terminada"
			&GridBadge.TooltipText = "Terminada"
			&GridBadge.Class = !' AttributeTagSuccess BootstrapTooltipRight'
		Otherwise
			&GridBadge = ''
			&GridBadge.TooltipText = ''
			&GridBadge.Class = !''
	EndCase
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

&AuditTableName = !'Turno'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'turno_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'turno_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'turno_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Turno.Load(&TurnoId)
			&LoadSuccess = &Turno.Success()
			If not &LoadSuccess
				&Messages = &Turno.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Turno_TurnoEnum.Enabled = False
				Turno_TurnoNombre.Enabled = False
				Turno_TurnoHoraInicio.Enabled = False
				Turno_TurnoHoraFin.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Do 'AttributesSecurityCode'
	Turno_TurnoId.Visible = False
	wcAuditoria.Object = AuditHistoryById.Create(&AuditTableName, &TurnoId)
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
	If &TurnoId > 0
		&Turno.TurnoEnum = EnumTurno.Convert(&TurnoId)
	EndIf

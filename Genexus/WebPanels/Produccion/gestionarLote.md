### Start (Event)

```genexus
&AuditTableName = !'Lote'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'lote_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'lote_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'lote_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Lote.Load(&LoteId)
			&LoadSuccess = &Lote.Success()
			If not &LoadSuccess
				&Messages = &Lote.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Lote_LoteSiloId.Enabled = False
				Lote_LoteEmbarque.Enabled = False
				Lote_LoteConsumido.Enabled = False
				Lote_LoteFechaRegistro.Enabled = False
				Lote_LoteKg.Enabled = False
				Lote_LotePaqueteAditivos.Enabled = False
				Lote_LoteTipoMaterial.Enabled = False
				Silo_SiloEstadoMaterial.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Do 'AttributesSecurityCode'
	Lote_LoteId.Visible = False
	wcAuditoria.Object = AuditHistoryById.Create(&AuditTableName, &LoteId)
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
	Do 'ActualizarDatosSilo'
```

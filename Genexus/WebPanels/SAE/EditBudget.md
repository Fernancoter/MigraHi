### 'DoRefreshData' (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
if not (&BudgetYear.IsEmpty() and &ConsolidatedName.IsEmpty() and &ProductNumber.IsEmpty())
	
	&SDTBudget.Clear()
	For each
		Where DB.CustomerStatus = 'A'
		Where WWPBaseObjects.ConsolidatedName = &ConsolidatedName
		
		&ItemSDTBudget = new()
		&ItemSDTBudget.ConsolidatedName = &ConsolidatedName
		&ItemSDTBudget.Customer = Customer
		&ItemSDTBudget.CustomerName = CustomerName
		&ItemSDTBudget.CustomerShipping = Mail.CustomerShipping 
		&ItemSDTBudget.ProductNumber = &ProductNumber
		&ItemSDTBudget.BudgetYear = &BudgetYear
		
		
		
		For &i = 1 to 12
			&Budget.Load(Customer, &ProductNumber, &BudgetYear, &i)
			&ItemSDTBudget.BudgetMonth = &i	
			
			Do Case
				Case &i = 1
					&ItemSDTBudget.BudgetMonth1 =  &Budget.BudgetEstimated
				Case &i = 2
					&ItemSDTBudget.BudgetMonth2 =  &Budget.BudgetEstimated
				Case &i = 3
					&ItemSDTBudget.BudgetMonth3 =  &Budget.BudgetEstimated
				Case &i = 4
					&ItemSDTBudget.BudgetMonth4 =  &Budget.BudgetEstimated
				Case &i = 5
					&ItemSDTBudget.BudgetMonth5 =  &Budget.BudgetEstimated
				Case &i = 6
					&ItemSDTBudget.BudgetMonth6 =  &Budget.BudgetEstimated
				Case &i = 7
					&ItemSDTBudget.BudgetMonth7 =  &Budget.BudgetEstimated
				Case &i = 8
					&ItemSDTBudget.BudgetMonth8 =  &Budget.BudgetEstimated
				Case &i = 9
					&ItemSDTBudget.BudgetMonth9 =  &Budget.BudgetEstimated
				Case &i = 10
					&ItemSDTBudget.BudgetMonth10 =  &Budget.BudgetEstimated
				Case &i = 11
					&ItemSDTBudget.BudgetMonth11 =  &Budget.BudgetEstimated
				Case &i = 12
					&ItemSDTBudget.BudgetMonth12 =  &Budget.BudgetEstimated
					
				Otherwise
					
			Endcase		
			
			
		Endfor
		&SDTBudget.Add(&ItemSDTBudget)
	endfor
	DO 'CalcularTotal'
else
msg('Please select filter')
endif
```

### 'DoUpdateData' (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
For &ItemSDTBudget in &SDTBudget
			For &i = 1 to 12
				
				&Budget.Load(&ItemSDTBudget.Customer, &ItemSDTBudget.ProductNumber, &ItemSDTBudget.BudgetYear, &i)
				Do Case
					Case &i = 1
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth1
					Case &i = 2

					Case &i = 3

					Case &i = 4
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth4
					Case &i = 5
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth5
					Case &i = 6
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth6
					Case &i = 7
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth7
					Case &i = 8
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth8
					Case &i = 9
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth9
					Case &i = 10
						&Budget.BudgetEstimated = &ItemSDTBudget.BudgetMonth10
```

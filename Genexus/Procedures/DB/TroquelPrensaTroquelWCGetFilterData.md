	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.TroquelPrensaTroquelWCGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.TroquelPrensaTroquelWCGridState"))
	Else

		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"

		EndCase
	EndFor

Sub 'LoadPrensaNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFPrensaNombre = &SearchTxt
	&TFPrensaNombre_Sel.SetEmpty()
	For Each PrensaTroquel
		Order TroquelId, PrensaNombre
		Where TroquelId = &TroquelId
		Where (PrensaTroquelId.ToString() like !'%' + &FilterFullText) OR (PrensaId.ToString() like !'%' + &FilterFullText) OR (PrensaNombre like !'%' + &FilterFullText) When not &FilterFullText.IsEmpty()
		Where PrensaNombre like &TFPrensaNombre when &TFPrensaNombre_Sel.IsEmpty() AND (not &TFPrensaNombre.IsEmpty())
		Where PrensaNombre = &TFPrensaNombre_Sel when not &TFPrensaNombre_Sel.IsEmpty()

		&count = 0
		For Each PrensaTroquel
			Order TroquelId, PrensaNombre
			&count += 1
		EndFor
		If not PrensaNombre.IsEmpty()
			&Option = PrensaNombre
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

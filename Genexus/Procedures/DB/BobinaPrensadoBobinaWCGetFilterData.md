Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.BobinaPrensadoBobinaWCGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.BobinaPrensadoBobinaWCGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"DB.BobinaPrensadoBobinaWCGridState"))
	Endif

		Do Case

				&TFBobinaEstado_SelsJson = &GridStateFilterValue.Value
				&TFBobinaEstado_Sels.FromJson(&TFBobinaEstado_SelsJson)
			Case &GridStateFilterValue.Name = !"PARM_&BOBINAID"
				&BobinaId.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

Sub 'LoadBobinaNoSerieOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFBobinaNoSerie = &SearchTxt
	&TFBobinaNoSerie_Sel.SetEmpty()
	For Each PrensadoBobina
		Order BobinaId
		Where BobinaId = &BobinaId
		Where (BobinaNoSerie like !'%' + &FilterFullText) OR ("en proceso" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"En Proceso") OR ("en medición" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"En Medición") OR ("reposo" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Reposo") OR ("molino" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Molino") OR ("disponible" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Disponible") OR ("en prensado" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"En Prensado") OR ("pausada" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Pausada") OR ("desmontada" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Desmontada") OR ("transferida" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Transferida") OR ("rechazada" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Rechazada") OR ("consumida" like !'%' + &FilterFullText.ToLower() AND BobinaEstado = !"Consumida") OR (PrensadoBobinaCantCarrera.ToString() like !'%' + &FilterFullText) OR (PrensadoOperadorId.ToString() like !'%' + &FilterFullText) OR (PrensadoOperadorNombre like !'%' + &FilterFullText) OR (PrensadoPrensaId.ToString() like !'%' + &FilterFullText) OR (PrensadoPrensaNombre like !'%' + &FilterFullText) OR (PrensadoTurnoId.ToString() like !'%' + &FilterFullText) OR (PrensadoTurnoNombre like !'%' + &FilterFullText) OR (PrensadoProductoId.ToString() like !'%' + &FilterFullText) OR (PrensadoProductoNombre like !'%' + &FilterFullText) OR ("pcr" like !'%' + &FilterFullText.ToLower() AND PrensadoProductoTipoMaterial = !"PCR") OR ("dow" like !'%' + &FilterFullText.ToLower() AND PrensadoProductoTipoMaterial = !"DOW") OR ("pcr 100%" like !'%' + &FilterFullText.ToLower() AND PrensadoProductoTipoMaterial = !"PCR100") When not &FilterFullText.IsEmpty()
		Where BobinaNoSerie like &TFBobinaNoSerie when &TFBobinaNoSerie_Sel.IsEmpty() AND (not &TFBobinaNoSerie.IsEmpty())
		Where BobinaNoSerie = &TFBobinaNoSerie_Sel when not &TFBobinaNoSerie_Sel.IsEmpty()
		Where BobinaEstado in &TFBobinaEstado_Sels when &TFBobinaEstado_Sels.Count > 0

		&count = 0
		For Each PrensadoBobina
			Order BobinaId
			&count += 1
		EndFor
		If not BobinaNoSerie.IsEmpty()
			&Option = BobinaNoSerie
			
			&InsertIndex = 1
			Do while &InsertIndex <= &Options.Count AND &Options.Item(&InsertIndex) < &Option
				&InsertIndex = &InsertIndex + 1
			EndDo
			&Options.Add(&Option, &InsertIndex)
			&OptionIndexes.Add(trim(&count.ToFormattedString()), &InsertIndex)
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

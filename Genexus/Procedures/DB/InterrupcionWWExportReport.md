	If &Session.Get(!"DB.InterrupcionWWGridState") = !""

		&GridState.FromXml(&Session.Get(!"DB.InterrupcionWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy

	For &GridStateFilterValue in &GridState.FilterValues

Sub 'OpenDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&Random = Random() * 10000
	&Filename = !"EmbarquePalletWWExport-" + &Random.ToString().Trim() + !".xlsx"

	&ExcelDocument.Open(&Filename)
	Do 'CheckStatus'
	&ExcelDocument.Clear()

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "WWP_FullTextFilterDescription")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&FilterFullText)
	EndIf
	If not (&TFEmbarqueDetalleId.IsEmpty() AND &TFEmbarqueDetalleId_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Embarque Detalle Id")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Number = &TFEmbarqueDetalleId
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Number = &TFEmbarqueDetalleId_To
	EndIf
	If not (&TFEmbarquePalletNoPallet_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "No Pallet")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFEmbarquePalletNoPallet_Sel)
	Else
		If not (&TFEmbarquePalletNoPallet.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "No Pallet")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFEmbarquePalletNoPallet)
		EndIf
	EndIf
	If not (&TFEmbarqueId.IsEmpty() AND &TFEmbarqueId_To.IsEmpty())
		WWPBaseObjects.WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Embarque Id")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Number = &TFEmbarqueId
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Number = &TFEmbarqueId_To
	EndIf
	&CellRow += 2

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'DB.EmbarquePalletWWColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'DB.EmbarquePalletWWColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each EmbarquePallet
		order EmbarquePalletValido  when &OrderedBy = 1
		order EmbarqueDetalleId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (EmbarqueDetalleId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order EmbarquePalletNoPallet  when &OrderedBy = 3 AND &OrderedDsc = False
		order (EmbarquePalletNoPallet)  when &OrderedBy = 3 AND &OrderedDsc = True
		order EmbarqueId  when &OrderedBy = 4 AND &OrderedDsc = False
		order (EmbarqueId)  when &OrderedBy = 4 AND &OrderedDsc = True
		
		using EmbarquePalletWWDS(&FilterFullText, &TFEmbarqueDetalleId, &TFEmbarqueDetalleId_To, &TFEmbarquePalletNoPallet, &TFEmbarquePalletNoPallet_Sel, &TFEmbarqueId
					, &TFEmbarqueId_To)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'EmbarqueDetalleId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = EmbarqueDetalleId
					Case &ColumnsSelector_Column.ColumnName = !'EmbarquePalletNoPallet'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(EmbarquePalletNoPallet)
					Case &ColumnsSelector_Column.ColumnName = !'EmbarqueId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = EmbarqueId
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"EmbarqueDetalleId", '', !'Embarque Detalle Id', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"EmbarquePalletNoPallet", '', !'No Pallet', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"EmbarqueId", '', !'Embarque Id', True, '')
		
	&UserCustomValue = LoadColumnsSelectorState.Udp(!'DB.EmbarquePalletWWColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.EmbarquePalletWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.EmbarquePalletWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"DB.EmbarquePalletWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEDETALLEID"
				&TFEmbarqueDetalleId.FromString(&GridStateFilterValue.Value)
				&TFEmbarqueDetalleId_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETNOPALLET"
				&TFEmbarquePalletNoPallet.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETNOPALLET_SEL"
				&TFEmbarquePalletNoPallet_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEID"
				&TFEmbarqueId.FromString(&GridStateFilterValue.Value)
				&TFEmbarqueId_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

EndSub

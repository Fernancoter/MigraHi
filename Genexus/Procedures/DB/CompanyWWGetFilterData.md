Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.CompanyWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.CompanyWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"DB.CompanyWWGridState"))
	Endif	

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCOMPANYID"
				&TFCompanyId.FromString(&GridStateFilterValue.Value)
				&TFCompanyId_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFCOMPANYNAME"
				&TFCompanyName.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCOMPANYNAME_SEL"
				&TFCompanyName_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCOMPANYCREATEDATETIME"
				&TFCompanyCreateDateTime.FromString(&GridStateFilterValue.Value)
				&TFCompanyCreateDateTime_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadCompanyNameOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFCompanyName = &SearchTxt
	&TFCompanyName_Sel.SetEmpty()
	For Each Company
		Order CompanyName
		using CompanyWWDS(&FilterFullText, &TFCompanyId, &TFCompanyId_To, &TFCompanyName, &TFCompanyName_Sel, &TFCompanyCreateDateTime
					, &TFCompanyCreateDateTime_To)
		Where CompanyId = &WWPContext.CompanyId when not &WWPContext.CompanyId.IsEmpty()

		&count = 0
		For Each Company
			Order CompanyName
			&count += 1
		EndFor
		If not CompanyName.IsEmpty()
			&Option = CompanyName
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

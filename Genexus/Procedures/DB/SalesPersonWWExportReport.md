Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each SalesPerson
		order SalesPersonActive  when &OrderedBy = 1 AND &OrderedDsc = False
		order (SalesPersonActive)  when &OrderedBy = 1 AND &OrderedDsc = True
		order SalesPersonName  when &OrderedBy = 2 AND &OrderedDsc = False
		order (SalesPersonName)  when &OrderedBy = 2 AND &OrderedDsc = True

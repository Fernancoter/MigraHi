Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Existencia
		order ExistenciaFechaHora  when &OrderedBy = 1 AND &OrderedDsc = False
		order (ExistenciaFechaHora)  when &OrderedBy = 1 AND &OrderedDsc = True
		order ExistenciaTurnoNombre  when &OrderedBy = 2 AND &OrderedDsc = False
		order (ExistenciaTurnoNombre)  when &OrderedBy = 2 AND &OrderedDsc = True
		
		using ExistenciaWWDS(&FilterFullText, &TFExistenciaFechaHora, &TFExistenciaFechaHora_To, &TFExistenciaTurnoNombre, &TFExistenciaTurnoNombre_Sel)

		Do 'BeforePrintLine'

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.ExistenciaWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.ExistenciaWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"DB.ExistenciaWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXISTENCIAFECHAHORA"
				&TFExistenciaFechaHora.FromString(&GridStateFilterValue.Value)
				&TFExistenciaFechaHora_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEXISTENCIATURNONOMBRE"
				&TFExistenciaTurnoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXISTENCIATURNONOMBRE_SEL"
				&TFExistenciaTurnoNombre_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&Random = Random() * 10000
	&Filename = !"listarSilosExport-" + &Random.ToString().Trim() + !".xlsx"

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())

	If not (&TFSiloNombre_Sel.IsEmpty())

		If not (&TFSiloNombre.IsEmpty())

			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFSiloNombre)
		EndIf
	EndIf
	If not (&TFSiloCapacidad.IsEmpty() AND &TFSiloCapacidad_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Capacidad (kg)")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Number = &TFSiloCapacidad
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Number = &TFSiloCapacidad_To
	EndIf
	If not (&TFSiloKgMinimo.IsEmpty() AND &TFSiloKgMinimo_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Mínimo (kg)")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Number = &TFSiloKgMinimo
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Number = &TFSiloKgMinimo_To
	EndIf
	If not (&TFSiloKgMaximo.IsEmpty() AND &TFSiloKgMaximo_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Máximo (kg)")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Number = &TFSiloKgMaximo
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Number = &TFSiloKgMaximo_To
	EndIf
	If not (&TFSiloEstadoMaterial_Sels.Count = 0)
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Estado de Material")
		&i = 1
		For &TFSiloEstadoMaterial_Sel in &TFSiloEstadoMaterial_Sels
			If &i = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = ''
			Else
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += ', '
			EndIf
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += &TFSiloEstadoMaterial_Sel.EnumerationDescription()
			&i += 1
		EndFor
	EndIf
	If not (&TFSiloTipoMaterial_Sels.Count = 0)
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Tipo de Material")
		&i = 1
		For &TFSiloTipoMaterial_Sel in &TFSiloTipoMaterial_Sels
			If &i = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = ''
			Else
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += ', '
			EndIf
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += &TFSiloTipoMaterial_Sel.EnumerationDescription()
			&i += 1
		EndFor
	EndIf
	If not (&TFSiloActivo_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Silo Activo")
		Do Case
			Case &TFSiloActivo_Sel = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSChecked"
			Case &TFSiloActivo_Sel = 2
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSUnChecked"
		EndCase
	EndIf

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'Produccion.listarSilosColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'Produccion.listarSilosColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Silo
		order SiloNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (SiloNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order SiloCapacidad  when &OrderedBy = 2 AND &OrderedDsc = False
		order (SiloCapacidad)  when &OrderedBy = 2 AND &OrderedDsc = True
		order SiloKgMinimo  when &OrderedBy = 3 AND &OrderedDsc = False
		order (SiloKgMinimo)  when &OrderedBy = 3 AND &OrderedDsc = True
		order SiloKgMaximo  when &OrderedBy = 4 AND &OrderedDsc = False
		order (SiloKgMaximo)  when &OrderedBy = 4 AND &OrderedDsc = True
		order SiloEstadoMaterial  when &OrderedBy = 5 AND &OrderedDsc = False
		order (SiloEstadoMaterial)  when &OrderedBy = 5 AND &OrderedDsc = True
		order SiloTipoMaterial  when &OrderedBy = 6 AND &OrderedDsc = False
		order (SiloTipoMaterial)  when &OrderedBy = 6 AND &OrderedDsc = True
		order SiloActivo  when &OrderedBy = 7 AND &OrderedDsc = False
		order (SiloActivo)  when &OrderedBy = 7 AND &OrderedDsc = True
		
		using listarSilosDS(&FilterFullText, &TFSiloNombre, &TFSiloNombre_Sel, &TFSiloCapacidad, &TFSiloCapacidad_To, &TFSiloKgMinimo
					, &TFSiloKgMinimo_To, &TFSiloKgMaximo, &TFSiloKgMaximo_To, &TFSiloEstadoMaterial_Sels, &TFSiloTipoMaterial_Sels, &TFSiloActivo_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'SiloNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(SiloNombre)
					Case &ColumnsSelector_Column.ColumnName = !'SiloCapacidad'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = SiloCapacidad
					Case &ColumnsSelector_Column.ColumnName = !'SiloKgMinimo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = SiloKgMinimo
					Case &ColumnsSelector_Column.ColumnName = !'SiloKgMaximo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = SiloKgMaximo
					Case &ColumnsSelector_Column.ColumnName = !'SiloEstadoMaterial'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = SiloEstadoMaterial.EnumerationDescription()
					Case &ColumnsSelector_Column.ColumnName = !'SiloTipoMaterial'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = SiloTipoMaterial.EnumerationDescription()
					Case &ColumnsSelector_Column.ColumnName = !'SiloActivo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = SiloActivo.ToString()
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

EndSub

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloNombre", '', !'Nombre', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloCapacidad", '', !'Capacidad (kg)', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloKgMinimo", '', !'Mínimo (kg)', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloKgMaximo", '', !'Máximo (kg)', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloEstadoMaterial", '', !'Estado de Material', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloTipoMaterial", '', !'Tipo de Material', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"SiloActivo", '', !'Silo Activo', True, '')
		
	&UserCustomValue = LoadColumnsSelectorState.Udp(!'Produccion.listarSilosColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarSilosGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarSilosGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarSilosGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILONOMBRE"
				&TFSiloNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILONOMBRE_SEL"
				&TFSiloNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILOCAPACIDAD"
				&TFSiloCapacidad.FromString(&GridStateFilterValue.Value)
				&TFSiloCapacidad_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOKGMINIMO"
				&TFSiloKgMinimo.FromString(&GridStateFilterValue.Value)
				&TFSiloKgMinimo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOKGMAXIMO"
				&TFSiloKgMaximo.FromString(&GridStateFilterValue.Value)
				&TFSiloKgMaximo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOESTADOMATERIAL_SEL"
				&TFSiloEstadoMaterial_SelsJson = &GridStateFilterValue.Value
				&TFSiloEstadoMaterial_Sels.FromJson(&TFSiloEstadoMaterial_SelsJson)
			Case &GridStateFilterValue.Name = !"TFSILOTIPOMATERIAL_SEL"
				&TFSiloTipoMaterial_SelsJson = &GridStateFilterValue.Value
				&TFSiloTipoMaterial_Sels.FromJson(&TFSiloTipoMaterial_SelsJson)
			Case &GridStateFilterValue.Name = !"TFSILOACTIVO_SEL"
				&TFSiloActivo_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

EndSub

	For each DB.Producto
		order ProductoNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (ProductoNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order ProductoCategoriaNombre  when &OrderedBy = 2 AND &OrderedDsc = False

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarProductosGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarProductosGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarProductosGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOCATEGORIANOMBRE"
				&TFProductoCategoriaNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOCATEGORIANOMBRE_SEL"
				&TFProductoCategoriaNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOBASEPRODUCTOCLAVE"
				&TFProductoBaseProductoClave.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOBASEPRODUCTOCLAVE_SEL"
				&TFProductoBaseProductoClave_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOCLAVE"
				&TFProductoClave.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOCLAVE_SEL"
				&TFProductoClave_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTONOMBRE"
				&TFProductoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTONOMBRE_SEL"
				&TFProductoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTODESCRIPCION"
				&TFProductoDescripcion.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTODESCRIPCION_SEL"
				&TFProductoDescripcion_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOPRECIOUNITARIO"
				&TFProductoPrecioUnitario.FromString(&GridStateFilterValue.Value)
				&TFProductoPrecioUnitario_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOINVENTARIOACTUAL"
				&TFProductoInventarioActual.FromString(&GridStateFilterValue.Value)
				&TFProductoInventarioActual_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOTIPOMATERIAL_SEL"
				&TFProductoTipoMaterial_SelsJson = &GridStateFilterValue.Value
				&TFProductoTipoMaterial_Sels.FromJson(&TFProductoTipoMaterial_SelsJson)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOACTIVO_SEL"
				&TFProductoActivo_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTOCLAVEEXTERNA"
				&TFProductoClaveExterna.FromString(&GridStateFilterValue.Value)
				&TFProductoClaveExterna_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRODUCTNUMBER"
				&TFProductNumber.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRODUCTNUMBER_SEL"
				&TFProductNumber_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

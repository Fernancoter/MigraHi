	For each DB.PrensaProducto
		order PrensaProductoId  when &OrderedBy = 1

		order (PrensaNombre)  when &OrderedBy = 2 AND &OrderedDsc = True

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarPrensaProductoGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarPrensaProductoGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarPrensaProductoGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues

			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"

			Case &GridStateFilterValue.Name = !"TFINSUMOPRODUCTONOMBRE_SEL"
				&TFInsumoProductoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCOMERCIALPRODUCTONOMBRE"
				&TFComercialProductoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCOMERCIALPRODUCTONOMBRE_SEL"

Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Silo
		order SiloNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (SiloNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order SiloCapacidad  when &OrderedBy = 2 AND &OrderedDsc = False
		order (SiloCapacidad)  when &OrderedBy = 2 AND &OrderedDsc = True
		order SiloKgMinimo  when &OrderedBy = 3 AND &OrderedDsc = False
		order (SiloKgMinimo)  when &OrderedBy = 3 AND &OrderedDsc = True
		order SiloKgMaximo  when &OrderedBy = 4 AND &OrderedDsc = False
		order (SiloKgMaximo)  when &OrderedBy = 4 AND &OrderedDsc = True
		order SiloEstadoMaterial  when &OrderedBy = 5 AND &OrderedDsc = False
		order (SiloEstadoMaterial)  when &OrderedBy = 5 AND &OrderedDsc = True
		order SiloTipoMaterial  when &OrderedBy = 6 AND &OrderedDsc = False
		order (SiloTipoMaterial)  when &OrderedBy = 6 AND &OrderedDsc = True
		order SiloActivo  when &OrderedBy = 7 AND &OrderedDsc = False
		order (SiloActivo)  when &OrderedBy = 7 AND &OrderedDsc = True
		
		using listarSilosDS(&FilterFullText, &TFSiloNombre, &TFSiloNombre_Sel, &TFSiloCapacidad, &TFSiloCapacidad_To, &TFSiloKgMinimo
					, &TFSiloKgMinimo_To, &TFSiloKgMaximo, &TFSiloKgMaximo_To, &TFSiloEstadoMaterial_Sels, &TFSiloTipoMaterial_Sels, &TFSiloActivo_Sel)
		&SiloEstadoMaterialDescription = SiloEstadoMaterial.EnumerationDescription()
		&SiloTipoMaterialDescription = SiloTipoMaterial.EnumerationDescription()

		Do 'BeforePrintLine'
		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarSilosGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarSilosGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarSilosGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILONOMBRE"
				&TFSiloNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILONOMBRE_SEL"
				&TFSiloNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFSILOCAPACIDAD"
				&TFSiloCapacidad.FromString(&GridStateFilterValue.Value)
				&TFSiloCapacidad_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOKGMINIMO"
				&TFSiloKgMinimo.FromString(&GridStateFilterValue.Value)
				&TFSiloKgMinimo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOKGMAXIMO"
				&TFSiloKgMaximo.FromString(&GridStateFilterValue.Value)
				&TFSiloKgMaximo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFSILOESTADOMATERIAL_SEL"
				&TFSiloEstadoMaterial_SelsJson = &GridStateFilterValue.Value
				&TFSiloEstadoMaterial_Sels.FromJson(&TFSiloEstadoMaterial_SelsJson)
			Case &GridStateFilterValue.Name = !"TFSILOTIPOMATERIAL_SEL"
				&TFSiloTipoMaterial_SelsJson = &GridStateFilterValue.Value
				&TFSiloTipoMaterial_Sels.FromJson(&TFSiloTipoMaterial_SelsJson)
			Case &GridStateFilterValue.Name = !"TFSILOACTIVO_SEL"
				&TFSiloActivo_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

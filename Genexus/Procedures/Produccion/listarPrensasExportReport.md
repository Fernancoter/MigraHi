Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Prensa
		order PrensaNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (PrensaNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order PrensaMarca  when &OrderedBy = 2 AND &OrderedDsc = False
		order (PrensaMarca)  when &OrderedBy = 2 AND &OrderedDsc = True
		order PrensaModelo  when &OrderedBy = 3 AND &OrderedDsc = False
		order (PrensaModelo)  when &OrderedBy = 3 AND &OrderedDsc = True
		
		using listarPrensasDS(&FilterFullText, &TFPrensaNombre, &TFPrensaNombre_Sel, &TFPrensaMarca, &TFPrensaMarca_Sel, &TFPrensaModelo
					, &TFPrensaModelo_Sel)

		Do 'BeforePrintLine'
		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarPrensasGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarPrensasGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarPrensasGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSANOMBRE"
				&TFPrensaNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSANOMBRE_SEL"
				&TFPrensaNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAMARCA"
				&TFPrensaMarca.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAMARCA_SEL"
				&TFPrensaMarca_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAMODELO"
				&TFPrensaModelo.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAMODELO_SEL"
				&TFPrensaModelo_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

EndSub

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

Sub 'LoadOperadorNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFOperadorNombre = &SearchTxt
	&TFOperadorNombre_Sel.SetEmpty()
	For Each DB.Operador
		Order DB.OperadorNombre
		using listarOperadorDS(&FilterFullText, &TFOperadorNombre, &TFOperadorNombre_Sel, &TFOperadorActivo_Sel)

		&count = 0
		For Each DB.Operador
			Order OperadorNombre
			&count += 1
		EndFor
		If not OperadorNombre.IsEmpty()
			&Option = OperadorNombre
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarExtrusoraGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarExtrusoraGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarExtrusoraGridState"))
	Endif	

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORANOMBRE"
				&TFExtrusoraNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORANOMBRE_SEL"
				&TFExtrusoraNombre_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadExtrusoraNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFExtrusoraNombre = &SearchTxt
	&TFExtrusoraNombre_Sel.SetEmpty()
	For Each DB.Extrusora
		Order ExtrusoraNombre
		using listarExtrusoraDS(&FilterFullText, &TFExtrusoraNombre, &TFExtrusoraNombre_Sel)

		&count = 0
		For Each DB.Extrusora
			Order DB.ExtrusoraNombre
			&count += 1
		EndFor
		If not ExtrusoraNombre.IsEmpty()
			&Option = ExtrusoraNombre
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

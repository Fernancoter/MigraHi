	&VisibleColumnCount = 0

			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))

			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.PrensaProducto
		order PrensaProductoId  when &OrderedBy = 1
		order PrensaNombre  when &OrderedBy = 2 AND &OrderedDsc = False
		order (PrensaNombre)  when &OrderedBy = 2 AND &OrderedDsc = True
		order InsumoProductoNombre  when &OrderedBy = 3 AND &OrderedDsc = False
		order (InsumoProductoNombre)  when &OrderedBy = 3 AND &OrderedDsc = True
		order ComercialProductoNombre  when &OrderedBy = 4 AND &OrderedDsc = False
		order (ComercialProductoNombre)  when &OrderedBy = 4 AND &OrderedDsc = True
		
		using Produccion.listarPrensaProductoDS(&FilterFullText, &TFPrensaNombre, &TFPrensaNombre_Sel, &TFInsumoProductoNombre, &TFInsumoProductoNombre_Sel, &TFComercialProductoNombre
					, &TFComercialProductoNombre_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'PrensaNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(PrensaNombre)
					Case &ColumnsSelector_Column.ColumnName = !'InsumoProductoNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(InsumoProductoNombre)
					Case &ColumnsSelector_Column.ColumnName = !'ComercialProductoNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(ComercialProductoNombre)
				EndCase

	&Random = Random() * 10000
	&Filename = !"listarTroquelExport-" + &Random.ToString().Trim() + !".xlsx"

	&ExcelDocument.Open(&Filename)
	Do 'CheckStatus'
	&ExcelDocument.Clear()

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "WWP_FullTextFilterDescription")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&FilterFullText)
	EndIf
	If not (&TFTroquelNombre_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Nombre")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFTroquelNombre_Sel)
	Else
		If not (&TFTroquelNombre.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Nombre")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFTroquelNombre)
		EndIf
	EndIf
	If not (&TFTroquelEnPrensa_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "En Prensa")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFTroquelEnPrensa_Sel)
	Else
		If not (&TFTroquelEnPrensa.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "En Prensa")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFTroquelEnPrensa)
		EndIf
	EndIf
	If not (&TFTroquelEstado_Sels.Count = 0)
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Estado")
		&i = 1
		For &TFTroquelEstado_Sel in &TFTroquelEstado_Sels
			If &i = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = ''
			Else
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += ', '
			EndIf
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text += &TFTroquelEstado_Sel.EnumerationDescription()
			&i += 1
		EndFor
	EndIf
	If not (&TFTroquelActivo_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Activo")
		Do Case
			Case &TFTroquelActivo_Sel = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSChecked"
			Case &TFTroquelActivo_Sel = 2
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSUnChecked"
		EndCase
	EndIf
	&CellRow += 2

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'Produccion.listarTroquelColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'Produccion.listarTroquelColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Troquel
		order TroquelNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (TroquelNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order TroquelEstado  when &OrderedBy = 2 AND &OrderedDsc = False
		order (TroquelEstado)  when &OrderedBy = 2 AND &OrderedDsc = True
		order TroquelActivo  when &OrderedBy = 3 AND &OrderedDsc = False
		order (TroquelActivo)  when &OrderedBy = 3 AND &OrderedDsc = True
		
		using listarTroquelDS(&FilterFullText, &TFTroquelNombre, &TFTroquelNombre_Sel, &TFTroquelEnPrensa, &TFTroquelEnPrensa_Sel, &TFTroquelEstado_Sels
					, &TFTroquelActivo_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'TroquelNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(TroquelNombre)
					Case &ColumnsSelector_Column.ColumnName = !'TroquelEnPrensa'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(TroquelEnPrensa)
					Case &ColumnsSelector_Column.ColumnName = !'TroquelEstado'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = TroquelEstado.EnumerationDescription()
					Case &ColumnsSelector_Column.ColumnName = !'TroquelActivo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = TroquelActivo.ToString()
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If (&ExcelDocument.ErrCode <> 0)
		&Filename = ""

		&ExcelDocument.Close()

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"TroquelNombre", '', !'Nombre', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"TroquelEnPrensa", '', !'En Prensa', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"TroquelEstado", '', !'Estado', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"TroquelActivo", '', !'Activo', True, '')
		
	&UserCustomValue = LoadColumnsSelectorState.Udp(!'Produccion.listarTroquelColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarTroquelGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarTroquelGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarTroquelGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTROQUELNOMBRE"
				&TFTroquelNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTROQUELNOMBRE_SEL"
				&TFTroquelNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTROQUELENPRENSA"
				&TFTroquelEnPrensa.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTROQUELENPRENSA_SEL"
				&TFTroquelEnPrensa_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTROQUELESTADO_SEL"
				&TFTroquelEstado_SelsJson = &GridStateFilterValue.Value
				&TFTroquelEstado_Sels.FromJson(&TFTroquelEstado_SelsJson)
			Case &GridStateFilterValue.Name = !"TFTROQUELACTIVO_SEL"
				&TFTroquelActivo_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

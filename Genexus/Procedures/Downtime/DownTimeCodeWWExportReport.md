Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DownTimeCode
		order DownTimeCodeName  when &OrderedBy = 1 AND &OrderedDsc = False
		order (DownTimeCodeName)  when &OrderedBy = 1 AND &OrderedDsc = True
		order DownTimeCodeId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (DownTimeCodeId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order DownTimeCodeDescription  when &OrderedBy = 3 AND &OrderedDsc = False
		order (DownTimeCodeDescription)  when &OrderedBy = 3 AND &OrderedDsc = True
		
		using DownTimeCodeWWDS(&FilterFullText, &TFDownTimeCodeId, &TFDownTimeCodeId_Sel, &TFDownTimeCodeName, &TFDownTimeCodeName_Sel, &TFDownTimeCodeDescription
					, &TFDownTimeCodeDescription_Sel)

		Do 'BeforePrintLine'
		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Downtime.DownTimeCodeWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Downtime.DownTimeCodeWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Downtime.DownTimeCodeWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEID"
				&TFDownTimeCodeId.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEID_SEL"
				&TFDownTimeCodeId_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODENAME"
				&TFDownTimeCodeName.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODENAME_SEL"
				&TFDownTimeCodeName_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEDESCRIPTION"
				&TFDownTimeCodeDescription.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEDESCRIPTION_SEL"
				&TFDownTimeCodeDescription_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

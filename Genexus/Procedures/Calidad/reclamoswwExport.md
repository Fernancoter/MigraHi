Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each Reclamo
		order ReclamoFecha  when &OrderedBy = 1 AND &OrderedDsc = False
		order (ReclamoFecha)  when &OrderedBy = 1 AND &OrderedDsc = True
		order ReclamoCodigo  when &OrderedBy = 2 AND &OrderedDsc = False
		order (ReclamoCodigo)  when &OrderedBy = 2 AND &OrderedDsc = True
		order ReclamoCliente  when &OrderedBy = 3 AND &OrderedDsc = False
		order (ReclamoCliente)  when &OrderedBy = 3 AND &OrderedDsc = True
		order PlantaClienteNombre  when &OrderedBy = 4 AND &OrderedDsc = False
		order (PlantaClienteNombre)  when &OrderedBy = 4 AND &OrderedDsc = True
		order ReclamoDescripcion  when &OrderedBy = 5 AND &OrderedDsc = False
		order (ReclamoDescripcion)  when &OrderedBy = 5 AND &OrderedDsc = True
		order ReclamoCarretesReportados  when &OrderedBy = 6 AND &OrderedDsc = False
		order (ReclamoCarretesReportados)  when &OrderedBy = 6 AND &OrderedDsc = True
		order ReclamoMillaresReportados  when &OrderedBy = 7 AND &OrderedDsc = False
		order (ReclamoMillaresReportados)  when &OrderedBy = 7 AND &OrderedDsc = True
		order ReclamoFechaCierre  when &OrderedBy = 8 AND &OrderedDsc = False
		order (SAE.ReclamoFechaCierre)  when &OrderedBy = 8 AND &OrderedDsc = True
		order ReclamoNotaCredito  when &OrderedBy = 9 AND &OrderedDsc = False
		order (ReclamoNotaCredito)  when &OrderedBy = 9 AND &OrderedDsc = True
		order ReclamoFechaNotaCredito  when &OrderedBy = 10 AND &OrderedDsc = False
		order (ReclamoFechaNotaCredito)  when &OrderedBy = 10 AND &OrderedDsc = True
		
		using reclamoswwDS(&FilterFullText, &TFReclamoFecha, &TFReclamoFecha_To, &TFReclamoCodigo, &TFReclamoCodigo_Sel, &TFReclamoCliente
					, &TFReclamoCliente_Sel, &TFPlantaClienteNombre, &TFPlantaClienteNombre_Sel, &TFReclamoDescripcion, &TFReclamoDescripcion_Sel, &TFReclamoCarretesReportados
					, &TFReclamoCarretesReportados_To, &TFReclamoMillaresReportados, &TFReclamoMillaresReportados_To, &TFReclamoFechaCierre, &TFReclamoFechaCierre_To, &TFReclamoNotaCredito
					, &TFReclamoNotaCredito_To, &TFReclamoFechaNotaCredito, &TFReclamoFechaNotaCredito_To)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'ReclamoFecha'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = ReclamoFecha
					Case &ColumnsSelector_Column.ColumnName = !'ReclamoCodigo'

					Case &ColumnsSelector_Column.ColumnName = !'ReclamoCliente'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(ReclamoCliente)
					Case &ColumnsSelector_Column.ColumnName = !'PlantaClienteNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(PlantaClienteNombre)
					Case &ColumnsSelector_Column.ColumnName = !'ReclamoDescripcion'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWPBaseObjects.WWP_Export_SecureText(ReclamoDescripcion)
					Case &ColumnsSelector_Column.ColumnName = !'ReclamoCarretesReportados'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = ReclamoCarretesReportados
					Case &ColumnsSelector_Column.ColumnName = !'ReclamoMillaresReportados'

Sub 'LoadGridState'

	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues

			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"

Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each ExtrusoraObservacion
		order ExtrusoraObservacionFecha  when &OrderedBy = 1 AND &OrderedDsc = False
		order (ExtrusoraObservacionFecha)  when &OrderedBy = 1 AND &OrderedDsc = True
		order ExtrusoraNombre  when &OrderedBy = 2 AND &OrderedDsc = False
		order (ExtrusoraNombre)  when &OrderedBy = 2 AND &OrderedDsc = True
		order TurnoNombre  when &OrderedBy = 3 AND &OrderedDsc = False
		order (TurnoNombre)  when &OrderedBy = 3 AND &OrderedDsc = True
		order ExtrusoraObservacionTiempoInterrupcion  when &OrderedBy = 4 AND &OrderedDsc = False
		order (ExtrusoraObservacionTiempoInterrupcion)  when &OrderedBy = 4 AND &OrderedDsc = True
		order ExtrusoraObservacionTiempoSupervisor  when &OrderedBy = 5 AND &OrderedDsc = False
		order (ExtrusoraObservacionTiempoSupervisor)  when &OrderedBy = 5 AND &OrderedDsc = True
		order ExtrusoraObservacionDescripcion  when &OrderedBy = 6 AND &OrderedDsc = False
		order (ExtrusoraObservacionDescripcion)  when &OrderedBy = 6 AND &OrderedDsc = True
		
		using ExtrusoraObservacionWWDS(&FilterFullText, &TFExtrusoraObservacionFecha, &TFExtrusoraObservacionFecha_To, &TFExtrusoraNombre, &TFExtrusoraNombre_Sel, &TFTurnoNombre

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Reportes.ExtrusoraObservacionWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Reportes.ExtrusoraObservacionWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Reportes.ExtrusoraObservacionWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORAOBSERVACIONFECHA"
				&TFExtrusoraObservacionFecha.FromString(&GridStateFilterValue.Value)
				&TFExtrusoraObservacionFecha_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORANOMBRE"
				&TFExtrusoraNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORANOMBRE_SEL"
				&TFExtrusoraNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE"
				&TFTurnoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE_SEL"
				&TFTurnoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORAOBSERVACIONTIEMPOINTERRUPCION"
				&TFExtrusoraObservacionTiempoInterrupcion.FromString(&GridStateFilterValue.Value)
				&TFExtrusoraObservacionTiempoInterrupcion_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORAOBSERVACIONTIEMPOSUPERVISOR"
				&TFExtrusoraObservacionTiempoSupervisor.FromString(&GridStateFilterValue.Value)
				&TFExtrusoraObservacionTiempoSupervisor_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORAOBSERVACIONDESCRIPCION"
				&TFExtrusoraObservacionDescripcion.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSORAOBSERVACIONDESCRIPCION_SEL"
				&TFExtrusoraObservacionDescripcion_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

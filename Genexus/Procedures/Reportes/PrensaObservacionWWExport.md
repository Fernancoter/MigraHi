Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each PrensaObservacion
		order PrensaObservacionFecha  when &OrderedBy = 1 AND &OrderedDsc = False
		order (PrensaObservacionFecha)  when &OrderedBy = 1 AND &OrderedDsc = True
		order PrensaNombre  when &OrderedBy = 2 AND &OrderedDsc = False
		order (PrensaNombre)  when &OrderedBy = 2 AND &OrderedDsc = True
		order TurnoNombre  when &OrderedBy = 3 AND &OrderedDsc = False
		order (TurnoNombre)  when &OrderedBy = 3 AND &OrderedDsc = True
		order PrensaObservacionTiempoInterrupcion  when &OrderedBy = 4 AND &OrderedDsc = False
		order (PrensaObservacionTiempoInterrupcion)  when &OrderedBy = 4 AND &OrderedDsc = True
		order PrensaObservacionTiempoSupervisor  when &OrderedBy = 5 AND &OrderedDsc = False
		order (PrensaObservacionTiempoSupervisor)  when &OrderedBy = 5 AND &OrderedDsc = True
		order PrensaObservacionDescripcion  when &OrderedBy = 6 AND &OrderedDsc = False
		order (PrensaObservacionDescripcion)  when &OrderedBy = 6 AND &OrderedDsc = True
		
		using PrensaObservacionWWDS(&FilterFullText, &TFPrensaObservacionFecha, &TFPrensaObservacionFecha_To, &TFPrensaNombre, &TFPrensaNombre_Sel, &TFTurnoNombre
					, &TFTurnoNombre_Sel, &TFPrensaObservacionTiempoInterrupcion, &TFPrensaObservacionTiempoInterrupcion_To, &TFPrensaObservacionTiempoSupervisor, &TFPrensaObservacionTiempoSupervisor_To, &TFPrensaObservacionDescripcion
					, &TFPrensaObservacionDescripcion_Sel)

		// Write cell values
		&CellRow += 1

		&VisibleColumnCount = 0

					Case &ColumnsSelector_Column.ColumnName = !'PrensaObservacionTiempoSupervisor'

					Case &ColumnsSelector_Column.ColumnName = !'PrensaObservacionDescripcion'

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Reportes.PrensaObservacionWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Reportes.PrensaObservacionWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Reportes.PrensaObservacionWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAOBSERVACIONFECHA"
				&TFPrensaObservacionFecha.FromString(&GridStateFilterValue.Value)
				&TFPrensaObservacionFecha_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRENSANOMBRE"
				&TFPrensaNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSANOMBRE_SEL"
				&TFPrensaNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE"
				&TFTurnoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE_SEL"
				&TFTurnoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAOBSERVACIONTIEMPOINTERRUPCION"
				&TFPrensaObservacionTiempoInterrupcion.FromString(&GridStateFilterValue.Value)
				&TFPrensaObservacionTiempoInterrupcion_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRENSAOBSERVACIONTIEMPOSUPERVISOR"
				&TFPrensaObservacionTiempoSupervisor.FromString(&GridStateFilterValue.Value)
				&TFPrensaObservacionTiempoSupervisor_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFPRENSAOBSERVACIONDESCRIPCION"
				&TFPrensaObservacionDescripcion.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPRENSAOBSERVACIONDESCRIPCION_SEL"
				&TFPrensaObservacionDescripcion_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

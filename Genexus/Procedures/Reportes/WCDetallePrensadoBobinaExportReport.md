Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.PrensadoBobina
		order BobinaNoSerie  when &OrderedBy = 1 AND &OrderedDsc = False
		order (BobinaNoSerie)  when &OrderedBy = 1 AND &OrderedDsc = True
		order BobinaEstado  when &OrderedBy = 2 AND &OrderedDsc = False
		order (BobinaEstado)  when &OrderedBy = 2 AND &OrderedDsc = True
		order BobinaNo  when &OrderedBy = 3 AND &OrderedDsc = False
		order (BobinaNo)  when &OrderedBy = 3 AND &OrderedDsc = True
		order BobinaOrigen  when &OrderedBy = 4 AND &OrderedDsc = False
		order (BobinaOrigen)  when &OrderedBy = 4 AND &OrderedDsc = True
		order ExtrusionExtrusoraNombre  when &OrderedBy = 5 AND &OrderedDsc = False
		order (ExtrusionExtrusoraNombre)  when &OrderedBy = 5 AND &OrderedDsc = True
		order BobinaKg  when &OrderedBy = 6 AND &OrderedDsc = False
		order (BobinaKg)  when &OrderedBy = 6 AND &OrderedDsc = True
		order BobinaHoraSalida  when &OrderedBy = 7 AND &OrderedDsc = False
		order (BobinaHoraSalida)  when &OrderedBy = 7 AND &OrderedDsc = True
		order BobinaCarreras  when &OrderedBy = 8 AND &OrderedDsc = False
		order (BobinaCarreras)  when &OrderedBy = 8 AND &OrderedDsc = True
		order BobinaMinutosEnReposo  when &OrderedBy = 9 AND &OrderedDsc = False
		order (BobinaMinutosEnReposo)  when &OrderedBy = 9 AND &OrderedDsc = True
		
		using WCDetallePrensadoBobinaDS(&FilterFullText, &TFBobinaNoSerie, &TFBobinaNoSerie_Sel, &TFBobinaEstado_Sels, &TFBobinaNo, &TFBobinaNo_To
					, &TFBobinaOrigen_Sels, &TFExtrusionExtrusoraNombre, &TFExtrusionExtrusoraNombre_Sel, &TFBobinaKg, &TFBobinaKg_To, &TFBobinaHoraSalida
					, &TFBobinaHoraSalida_To, &TFBobinaCarreras, &TFBobinaCarreras_To, &TFBobinaMinutosEnReposo, &TFBobinaMinutosEnReposo_To, &TFBobinaReposoEnHoras
					, &TFBobinaReposoEnHoras_To)
		Where PrensadoId = &PrensadoId when not &PrensadoId.IsEmpty()
		&BobinaEstadoDescription = BobinaEstado.EnumerationDescription()
		&BobinaOrigenDescription = BobinaOrigen.EnumerationDescription()

		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Reportes.WCDetallePrensadoBobinaGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Reportes.WCDetallePrensadoBobinaGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Reportes.WCDetallePrensadoBobinaGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFBOBINANOSERIE"
				&TFBobinaNoSerie.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFBOBINANOSERIE_SEL"
				&TFBobinaNoSerie_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFBOBINAESTADO_SEL"
				&TFBobinaEstado_SelsJson = &GridStateFilterValue.Value
				&TFBobinaEstado_Sels.FromJson(&TFBobinaEstado_SelsJson)
			Case &GridStateFilterValue.Name = !"TFBOBINANO"
				&TFBobinaNo.FromString(&GridStateFilterValue.Value)
				&TFBobinaNo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFBOBINAORIGEN_SEL"
				&TFBobinaOrigen_SelsJson = &GridStateFilterValue.Value
				&TFBobinaOrigen_Sels.FromJson(&TFBobinaOrigen_SelsJson)
			Case &GridStateFilterValue.Name = !"TFEXTRUSIONEXTRUSORANOMBRE"
				&TFExtrusionExtrusoraNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEXTRUSIONEXTRUSORANOMBRE_SEL"
				&TFExtrusionExtrusoraNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFBOBINAKG"
				&TFBobinaKg.FromString(&GridStateFilterValue.Value)
				&TFBobinaKg_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFBOBINAHORASALIDA"
				&TFBobinaHoraSalida.FromString(&GridStateFilterValue.Value)
				&TFBobinaHoraSalida_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFBOBINACARRERAS"
				&TFBobinaCarreras.FromString(&GridStateFilterValue.Value)
				&TFBobinaCarreras_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFBOBINAMINUTOSENREPOSO"
				&TFBobinaMinutosEnReposo.FromString(&GridStateFilterValue.Value)
				&TFBobinaMinutosEnReposo_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFBOBINAREPOSOENHORAS"
				&TFBobinaReposoEnHoras.FromString(&GridStateFilterValue.Value)
				&TFBobinaReposoEnHoras_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

EndSub

Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.EmbarquePallet
		order EmbarquePalletValido  when &OrderedBy = 1
		order EmbarqueFecha  when &OrderedBy = 2 AND &OrderedDsc = False
		order (EmbarqueFecha)  when &OrderedBy = 2 AND &OrderedDsc = True
		order EmbarqueCliente  when &OrderedBy = 3 AND &OrderedDsc = False
		order (EmbarqueCliente)  when &OrderedBy = 3 AND &OrderedDsc = True
		order EmbarqueClienteGrupo  when &OrderedBy = 4 AND &OrderedDsc = False
		order (EmbarqueClienteGrupo)  when &OrderedBy = 4 AND &OrderedDsc = True
		order EmbarqueClienteEnvia  when &OrderedBy = 5 AND &OrderedDsc = False
		order (EmbarqueClienteEnvia)  when &OrderedBy = 5 AND &OrderedDsc = True
		order EmbarqueDetalleProducto  when &OrderedBy = 6 AND &OrderedDsc = False
		order (EmbarqueDetalleProducto)  when &OrderedBy = 6 AND &OrderedDsc = True
		order EmbarquePalletNoPallet  when &OrderedBy = 7 AND &OrderedDsc = False
		order (EmbarquePalletNoPallet)  when &OrderedBy = 7 AND &OrderedDsc = True
		
		using PalletEnEmbarqueDS(&FilterFullText, &TFEmbarqueFecha, &TFEmbarqueFecha_To, &TFEmbarqueCodigo, &TFEmbarqueCodigo_Sel, &TFEmbarqueCliente

					, &TFEmbarqueDetalleProducto_Sel, &TFEmbarquePalletNoPallet, &TFEmbarquePalletNoPallet_Sel)
		Where EmbarqueEstatus in (EstatusEmbarque.Finalizado)

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Reportes.PalletEnEmbarqueGridState") = !""
		&GridState.FromXml(WWPBaseObjects.LoadGridState.Udp(!"Reportes.PalletEnEmbarqueGridState"))
	Else

	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEFECHA"

				&TFEmbarqueFecha_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEMBARQUECODIGO"
				&TFEmbarqueCodigo.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUECODIGO_SEL"
				&TFEmbarqueCodigo_Sel.FromString(&GridStateFilterValue.Value)

			Case &GridStateFilterValue.Name = !"TFEMBARQUECLIENTEGRUPO_SEL"
				&TFEmbarqueClienteGrupo_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUECLIENTEENVIA"

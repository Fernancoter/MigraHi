	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Embarques.EmbarqueDetallePalletWCGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Embarques.EmbarqueDetallePalletWCGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Embarques.EmbarqueDetallePalletWCGridState"))
	Endif	

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETNOPALLET"
				&TFEmbarquePalletNoPallet.FromString(&GridStateFilterValue.Value)

				&TFEmbarquePalletNoPallet_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETMOTIVOERROR"
				&TFEmbarquePalletMotivoError.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETMOTIVOERROR_SEL"
				&TFEmbarquePalletMotivoError_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

Sub 'LoadEmbarquePalletNoPalletOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFEmbarquePalletNoPallet = &SearchTxt
	&TFEmbarquePalletNoPallet_Sel.SetEmpty()
	For Each DB.EmbarquePallet
		Order EmbarquePalletNoPallet
		using EmbarqueDetallePalletWCDS(&TFEmbarquePalletNoPallet, &TFEmbarquePalletNoPallet_Sel, &TFEmbarquePalletMotivoError, &TFEmbarquePalletMotivoError_Sel)
		Where EmbarqueId = &EmbarqueId
		Where EmbarqueDetalleId = &EmbarqueDetalleId

		&count = 0
		For Each DB.EmbarquePallet
			Order EmbarquePalletNoPallet
			&count += 1
		EndFor
		If not EmbarquePalletNoPallet.IsEmpty()
			&Option = Mail.EmbarquePalletNoPallet
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

	EndFor

Sub 'Update Notification Definitions'

  	&WWP_NotificationDefinitionCollection = WWPBaseObjects.Notifications.Common.WWP_AutomaticNotificationDefinitionsToLoad()
	For &WWP_NotificationDefinition In &WWP_NotificationDefinitionCollection
		&WWPNotificationDefinitionId = WWP_GetNotificationDefinitionByName(&WWP_NotificationDefinition.WWPNotificationDefinitionName, &WWP_NotificationDefinition.WWPEntityId)
		If &WWPNotificationDefinitionId > 0
			Msg(format(!'Updating Notification %1...', &WWP_NotificationDefinition.WWPNotificationDefinitionName), status)
			&WWP_NotificationDefinitionToUpdate.Load(&WWPNotificationDefinitionId)
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionAllowUserSubscription = &WWP_NotificationDefinition.WWPNotificationDefinitionAllowUserSubscription
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionAppliesTo = &WWP_NotificationDefinition.WWPNotificationDefinitionAppliesTo
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionDescription = &WWP_NotificationDefinition.WWPNotificationDefinitionDescription
			&WWP_NotificationDefinitionToUpdate.WWPBaseObjects.WWPNotificationDefinitionIcon = &WWP_NotificationDefinition.WWPNotificationDefinitionIcon
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionLink = &WWP_NotificationDefinition.WWPNotificationDefinitionLink
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionLongDescription = &WWP_NotificationDefinition.WWPNotificationDefinitionLongDescription
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionShortDescription = &WWP_NotificationDefinition.WWPNotificationDefinitionShortDescription
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionTitle = &WWP_NotificationDefinition.WWPNotificationDefinitionTitle
			&WWP_NotificationDefinitionToUpdate.WWPNotificationDefinitionSecFuncionality = &WWP_NotificationDefinition.WWPNotificationDefinitionSecFuncionality
			&WWP_NotificationDefinitionToUpdate.Save()
			If &WWP_NotificationDefinitionToUpdate.Fail()
				Msg(!"Error: " + &WWP_NotificationDefinitionToUpdate.GetMessages().ToJson(), status)
			EndIf
		Else
		Msg(format(!'Creating Notification %1...', &WWP_NotificationDefinition.WWPBaseObjects.WWPNotificationDefinitionName), status)
			&WWP_NotificationDefinition.Save()
			If &WWP_NotificationDefinition.Fail()
				Msg(!"Error: " + &WWP_NotificationDefinition.GetMessages().ToJson(), status)
			EndIf
		EndIf
	EndFor

EndSub

Sub 'Import Mail Template'

	&MailTemplate.Load(!"MailNotification")
	If &MailTemplate.Fail()
		Msg(!'Importing Mail template...', status)
		&TemplateFile.Source = &MailTemplatePath
		
		&MailTemplate = new()
		&MailTemplate.WWPMailTemplateName = !"MailNotification"
		&MailTemplate.WWPMailTemplateDescription = !"Template for sending notifications"
			
		WWPBaseObjects.WWP_GetParameter.Text(!"Sender_Name", &SenderName)
		&MailTemplate.WWPMailTemplateSenderName = &SenderName
		
		WWP_GetParameter.Text(!"Sender_Address", &SenderAddress)
		&MailTemplate.WWPMailTemplateSenderAddress = &SenderAddress
    
		&MailTemplate.WWPMailTemplateSubject = !"New notification"
		&MailTemplate.WWPMailTemplateBody = &TemplateFile.ReadAllText()
		&MailTemplate.Save()	
	EndIf

EndSub

```genexus
	For Each WWP_Entity
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
		&WWPEntityName = WWPEntityName
		&WWPEntityId = WWPEntityId
		&EntityHasItemsToShow = False
		&NotificationsForEntityCount = Count(WWPNotificationDefinitionDescription, WWPNotificationDefinitionAppliesTo = Notifications.WWP_NotificationAppliesTo.AnyRecord)
		If &NotificationsForEntityCount > 0 And Not &NotifShowOnlySubscribedEvents
			&EntityHasItemsToShow = True
		Else
			&SubscriptionsForEntityCount = Count(WWPSubscriptionSubscribed, WWPSubscriptionRoleId = &WWPSubscriptionRoleId)
			If &SubscriptionsForEntityCount > 0
				&EntityHasItemsToShow = True
			EndIf
		EndIf
		If &EntityHasItemsToShow
			&GridRecordCount += 1
			Load
		EndIf
		
	EndFor
	&GridPageCount = &GridRecordCount/Grid.Rows + iif(Mod(&GridRecordCount, Grid.Rows) > 0, 1, 0)
```

### 'DoRefreshData' (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
&SDTFTB.Clear()
	
	For each
		
		Where InvoiceDate >= &StartInvoiceDate
		Where InvoiceDate <= &EndInvoiceDate
	
		&FTB.Load(Invoice, ProductNumber, NumPar)
		
		&ItemSDTFTB = new ()
		&ItemSDTFTB.Invoice= &FTB.Invoice
		&ItemSDTFTB.ProductNumber= &FTB.ProductNumber
		&ItemSDTFTB.NumPar= &FTB.NumPar
		&ItemSDTFTB.ConsolidatedName= &FTB.ConsolidatedName
		&ItemSDTFTB.InvoiceDate= &FTB.InvoiceDate
		&ItemSDTFTB.Quantity= &FTB.Quantity
		&ItemSDTFTB.FTBShipping = &FTB.FTBShipping
		&ItemSDTFTB.Customer = &FTB.FTBCustomer
		&ItemSDTFTB.BudgetCustomer = &FTB.BudgetCustomer
		
		&SDTFTB.Add(&ItemSDTFTB)
			
	endfor
```

### 'DoUpdateData' (Event)

```genexus
For &ItemSDTFTB in &SDTFTB
		&FTB.Load(&ItemSDTFTB.Invoice, &ItemSDTFTB.ProductNumber, &ItemSDTFTB.NumPar)
		if(not &ItemSDTFTB.BudgetCustomer.IsEmpty())
			&FTB.BudgetCustomer = &ItemSDTFTB.BudgetCustomer
			&FTB.Save()
			commit
		endif
	Endfor
	msg('Updated')
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

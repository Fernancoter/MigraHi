### DDO_Grid.OnOptionClicked (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	Do Case
		Case DDO_Grid.ActiveEventKey = !'<#OrderASC#>' OR DDO_Grid.ActiveEventKey = !'<#OrderDSC#>'
			&OrderedBy.FromString(DDO_Grid.SelectedValue_get)
			&OrderedDsc = iif(DDO_Grid.ActiveEventKey = !'<#OrderDSC#>', true, false)
			Do 'SetDDOSortedStatus'
			Grid.FirstPage()
		Case DDO_Grid.ActiveEventKey = !'<#Filter#>'
			Do Case
				Case DDO_Grid.SelectedColumn = !'ExtrusionExtrusoraNombre'
					&TFExtrusionExtrusoraNombre = DDO_Grid.FilteredText_get
					&TFExtrusionExtrusoraNombre_Sel = DDO_Grid.SelectedValue_get
				Case DDO_Grid.SelectedColumn = !'ExtrusionFecha'
					&TFExtrusionFecha.FromString(DDO_Grid.FilteredText_get)
					&TFExtrusionFecha_To.FromString(DDO_Grid.FilteredTextTo_get)
					If Not &TFExtrusionFecha_To.IsEmpty()
						&TFExtrusionFecha_To.Set(&TFExtrusionFecha_To.Year(), &TFExtrusionFecha_To.Month(), &TFExtrusionFecha_To.Day(), 23, 59, 59)
					EndIf
				Case DDO_Grid.SelectedColumn = !'BobinaNoSerie'
					&TFBobinaNoSerie = DDO_Grid.FilteredText_get
					&TFBobinaNoSerie_Sel = DDO_Grid.SelectedValue_get
				Case DDO_Grid.SelectedColumn = !'BobinaOrigen'
					&TFBobinaOrigen_SelsJson = DDO_Grid.SelectedValue_get
					&TFBobinaOrigen_Sels.FromJson(&TFBobinaOrigen_SelsJson)
				Case DDO_Grid.SelectedColumn = !'BobinaHoraInicio'
					&TFBobinaHoraInicio.FromString(DDO_Grid.FilteredText_get)
					&TFBobinaHoraInicio_To.FromString(DDO_Grid.FilteredTextTo_get)
					If Not &TFBobinaHoraInicio_To.IsEmpty()
						&TFBobinaHoraInicio_To.Set(&TFBobinaHoraInicio_To.Year(), &TFBobinaHoraInicio_To.Month(), &TFBobinaHoraInicio_To.Day(), 23, 59, 59)

					&TFBobinaHoraSalida.FromString(DDO_Grid.FilteredText_get)
					&TFBobinaHoraSalida_To.FromString(DDO_Grid.FilteredTextTo_get)
```

### Grid.Load (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&Update = !'<i class="fa fa-pen"></i>'
	If (&IsAuthorized_Update)
		&Update.Link = DB.EmbarqueDetalle.Link(TrnMode.Update, EmbarqueDetalleId)
	Endif
	&Delete = !'<i class="fa fa-times"></i>'
	If (&IsAuthorized_Delete)
		&Delete.Link = DB.EmbarqueDetalle.Link(TrnMode.Delete, EmbarqueDetalleId)
	Endif
	&GridActions.Clear()
	&GridActions.AddItem(0, !";fa fa-bars")
	If &IsAuthorized_CambiarFecha
		If (EmbarqueEstatus = EstatusEmbarque.Programado or EmbarqueEstatus = EstatusEmbarque.PorProgramar)
			&GridActions.AddItem(1, 'Actualizar Datos')
		Endif
	EndIf
	If (EmbarqueEstatus = EstatusEmbarque.Programado)
		&GridActions.AddItem(2, 'Cargar')
	Endif
	If &IsAuthorized_Continuar
		If (EmbarqueEstatus = EstatusEmbarque.EnProceso)
			&GridActions.AddItem(3, 'Continuar Cargando')
		Endif
	EndIf
	If &IsAuthorized_cmdCretificadoCalidad
		If (EmbarqueEstatus = EstatusEmbarque.Finalizado)
			&GridActions.AddItem(4, 'Cert. Calidad')
		Endif
	EndIf
	If &IsAuthorized_cmdValeSalida
		If (EmbarqueEstatus = EstatusEmbarque.Finalizado)
			&GridActions.AddItem(5, 'Vale Salida')
		Endif
	EndIf
	If &GridActions.Count = 1
		&GridActions.Class = !'Invisible'
	Else
		&GridActions.Class = !'ConvertToDDO'
	EndIf
	
	&DetailWebComponent = !'<i class="fas fa-info-circle"></i>'
	If (EmbarqueEstatus in (EstatusEmbarque.EnProceso, EstatusEmbarque.Finalizado))
		&DetailWebComponent.Class = !'Attribute'
	Else
		&DetailWebComponent.Class = !'Invisible'
	Endif
	&Display = !'<i class="fa fa-search"></i>'
	If (&IsAuthorized_Display)
		&Display.Link = DB.EmbarqueView.Link(EmbarqueId, &TabCode)
	Endif
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

### Grid.Load (Event)

/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&Update = !'<i class="fa fa-pen"></i>'
	If (&IsAuthorized_Update)

	Endif
	&Delete = !'<i class="fa fa-times"></i>'

	&DetailWebComponent = !'<i class="fas fa-info-circle"></i>'

		&DetailWebComponent.Class = !'Attribute'
	Else
		&DetailWebComponent.Class = !'Invisible'
	Endif
	If (&IsAuthorized_PrensadoFecha)
		PrensadoFecha.Link = DB.PrensadoView.Link(PrensadoId, '')
	Endif
	Do Case
		Case PrensadoEstado = EstadoPrensado.Programado
			&GridBadge = "Programado"
			&GridBadge.TooltipText = "Programado"
			&GridBadge.Class = !' AttributeTagInfo BootstrapTooltipRight'

			&GridBadge = "Terminado"
			&GridBadge.TooltipText = "Terminado"
			&GridBadge.Class = !' AttributeTagSuccess BootstrapTooltipRight'
		Otherwise
			&GridBadge = ''
			&GridBadge.TooltipText = ''
			&GridBadge.Class = !''
	EndCase

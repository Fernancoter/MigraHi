### Start (Event)

```genexus
&BobinaProductoCategoriaNombre = ObtenerConfiguracion.Udp(!'CategoriaBobina')
	&CarreteProductoCategoriaNombre = ObtenerConfiguracion.Udp(!'CategoriaCarrete')
	&AuditTableName = !'PrensaProducto'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'prensaproducto_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'prensaproducto_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'prensaproducto_Delete'))
		If &TrnMode <> TrnMode.Insert
			&PrensaProducto.Load(&PrensaProductoId)
			&LoadSuccess = &PrensaProducto.Success()
			If not &LoadSuccess
				&Messages = &PrensaProducto.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				PrensaProducto_PrensaId.Enabled = False
				Combo_PrensaProducto_ComercialProductoId.Enabled = False
				Combo_PrensaProducto_InsumoProductoId.Enabled = False
				&IsWithFilter.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	&DDO_TitleSettingsIcons = GetWWPTitleSettingsIcons()
	Do 'LoadComboPrensaProducto_InsumoProductoId'
	Do 'LoadComboPrensaProducto_ComercialProductoId'
	Do 'AttributesSecurityCode'
	PrensaProducto_PrensaProductoId.Visible = False
	PrensaProducto_PrensaNombre.Visible = False
	PrensaProducto_InsumoProductoNombre.Visible = False
	PrensaProducto_ComercialProductoNombre.Visible = False
	wcAuditoria.Object = WWPBaseObjects.AuditHistoryById.Create(&AuditTableName, &PrensaProductoId)
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

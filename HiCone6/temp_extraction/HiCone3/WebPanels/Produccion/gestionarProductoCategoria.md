&AuditTableName = !'ProductoCategoria'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'productocategoria_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'productocategoria_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'productocategoria_Delete'))
		If &TrnMode <> TrnMode.Insert
			&ProductoCategoria.Load(&ProductoCategoriaId)
			&LoadSuccess = &ProductoCategoria.Success()
			If not &LoadSuccess
				&Messages = &ProductoCategoria.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				ProductoCategoria_ProductoCategoriaNombre.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Do 'AttributesSecurityCode'
	ProductoCategoria_ProductoCategoriaId.Visible = False
	ProductoCategoria_ProductoCategoriaClaveExterna.Visible = False
	wcAuditoria.Object = AuditHistoryById.Create(&AuditTableName, &ProductoCategoriaId)
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

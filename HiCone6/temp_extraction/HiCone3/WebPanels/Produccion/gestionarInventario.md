### Start (Event)

```genexus
&AuditTableName = !'Inventario'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'inventario_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'inventario_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'inventario_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Inventario.Load(&InventarioId)
			&LoadSuccess = &Inventario.Success()
			If not &LoadSuccess
				&Messages = &Inventario.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Inventario_InventarioFechaHora.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Do 'AttributesSecurityCode'
	wcAuditoria.Object = AuditHistoryById.Create(&AuditTableName, &InventarioId)
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

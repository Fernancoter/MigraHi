&AuditTableName = !'Silo'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'silo_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'silo_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'silo_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Silo.Load(&SiloId)
			&LoadSuccess = &Silo.Success()
			If not &LoadSuccess
				&Messages = &Silo.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Silo_SiloActivo.Enabled = False
				Silo_SiloNombre.Enabled = False
				Silo_SiloCapacidad.Enabled = False
				Silo_SiloKgMinimo.Enabled = False
				Silo_SiloKgMaximo.Enabled = False
				Silo_SiloEstadoMaterial.Enabled = False
				Silo_SiloTipoMaterial.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Do 'AttributesSecurityCode'
	wcAuditoria.Object = AuditHistoryById.Create(&AuditTableName, &SiloId)

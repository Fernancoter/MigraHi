&ProductoCategoriaNombre = ObtenerConfiguracion.Udp(!'CategoriaBobina')
	&AuditTableName = !'ExtrusoraProducto'
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = GeneXus.TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'extrusoraproducto_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'extrusoraproducto_Update'))

			&ExtrusoraProducto.Load(&ExtrusoraProductoId)
			&LoadSuccess = &ExtrusoraProducto.Success()
			If not &LoadSuccess
				&Messages = &ExtrusoraProducto.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				ExtrusoraProducto_ExtrusoraId.Enabled = False
				Combo_ExtrusoraProducto_ProductoId.Enabled = False
				ExtrusoraProducto_ExtrusoraProductoCalibre.Enabled = False
				ExtrusoraProducto_ExtrusoraProductoAncho.Enabled = False

				ExtrusoraProducto_ExtrusoraProductoLongitud.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	&DDO_TitleSettingsIcons = GetWWPTitleSettingsIcons()
	Do 'LoadComboExtrusoraProducto_ProductoId'
	Do 'AttributesSecurityCode'

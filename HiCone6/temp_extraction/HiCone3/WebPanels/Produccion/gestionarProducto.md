### Start (Event)

```genexus
&AuditTableName = !'Producto'
	Do Case
		Case &TrnMode = TrnMode.Insert
			&Producto.ProductoActivo = True
	EndCase
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'producto_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'producto_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'producto_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Producto.Load(&ProductoId)
			&LoadSuccess = &Producto.Success()
			If not &LoadSuccess
				&Messages = &Producto.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Producto_ProductNumber.Enabled = False
				Producto_ProductoCategoriaId.Enabled = False
				Producto_ProductoClave.Enabled = False
				Producto_ProductoNombre.Enabled = False
				Producto_ProductoDescripcion.Enabled = False
				Producto_ProductoPrecioUnitario.Enabled = False
				Producto_ProductoInventarioActual.Enabled = False
				Producto_ProductoClaveExterna.Enabled = False
				Producto_ProductoTipoMaterial.Enabled = False
				Producto_ProductoImagen.Enabled = False
				Producto_ProductoActivo.Enabled = False
				&ProductoCategoriaId.Enabled = False
				Combo_ProductoBaseProductoId.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	&DDO_TitleSettingsIcons = GetWWPTitleSettingsIcons()
	&ProductoBaseProductoId.Visible = False
	Do 'LoadComboProductoBaseProductoId'
	Do 'AttributesSecurityCode'
	Producto_ProductoId.Visible = False
```

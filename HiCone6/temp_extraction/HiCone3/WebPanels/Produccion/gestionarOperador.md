### Start (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	&LoadSuccess = True
	If (&TrnMode = TrnMode.Display)
		OR (&TrnMode = TrnMode.Insert AND SecGAMIsAuthByFunctionalityKey.Udp(!'operador_Insert'))
		OR (&TrnMode = TrnMode.Update AND SecGAMIsAuthByFunctionalityKey.Udp(!'operador_Update'))
		OR (&TrnMode = TrnMode.Delete AND SecGAMIsAuthByFunctionalityKey.Udp(!'operador_Delete'))
		If &TrnMode <> TrnMode.Insert
			&Operador.Load(&OperadorId)
			&LoadSuccess = &Operador.Success()
			If not &LoadSuccess
				&Messages = &Operador.GetMessages()
				Do 'Show Messages'
			EndIf
			If &TrnMode = TrnMode.Display OR &TrnMode = TrnMode.Delete
				Operador_OperadorNombre.Enabled = False
				Operador_OperadorFotografia.Enabled = False
				Operador_OperadorActivo.Enabled = False
			EndIf
		EndIf
	Else
		&LoadSuccess = False
	EndIf
	If &LoadSuccess
		If &TrnMode = TrnMode.Delete
			msg('GXM_confdelete')
		EndIf
	EndIf
	Operador_OperadorId.Visible = False
	Operador_OperadorUserGUID.Visible = False
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

			&isOK = DeshabilitarOperador.Udp(&Operador)
			If &isOK
				Commit
				Return
			Else
				RollBack
			EndIf
		OtherWise
	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	If &TrnMode <> TrnMode.Display
		If &TrnMode <> TrnMode.Delete
			Do 'CheckRequiredFields'
		EndIf
		If &TrnMode = TrnMode.Delete OR &CheckRequiredFieldsResult
			If &TrnMode = TrnMode.Delete
				&Operador.Delete()
			Else
				&Operador.Save()
			EndIf
			If &Operador.Success()
				Do 'After_Trn'

			EndIf
		EndIf
	EndIf

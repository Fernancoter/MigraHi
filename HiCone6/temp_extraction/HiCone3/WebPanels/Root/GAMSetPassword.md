### Enter (Event)

```genexus
&User.Load(&UserId)
	
	If &UserPasswordNew = &UserPasswordNewConf 
		If &User.Success()
			&IsChanged = &User.SetPassword(&UserPasswordNew, &Errors)
			If &IsChanged
				Commit
				&UserPasswordNew.Enabled 	= False
				&UserPasswordNewConf.Enabled= False
				BtnEnter.Visible = False
				Msg("WWP_GAM_PasswordChangedSuccesfully")
				BtnCancel.Caption = "Back to Users"
			Else
				Do 'DisplayMessages'
			Endif
		Else
			&Errors = &User.GetErrors()
			Do 'DisplayMessages'
		Endif
	Else
		Msg("WWP_GAM_NewPasswordNotMatch")
	Endif
```

### DDO_Grid.OnOptionClicked (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	Do Case
		Case DDO_Grid.ActiveEventKey = !'<#OrderASC#>' OR DDO_Grid.ActiveEventKey = !'<#OrderDSC#>'
			&OrderedBy.FromString(DDO_Grid.SelectedValue_get)
			&OrderedDsc = iif(DDO_Grid.ActiveEventKey = !'<#OrderDSC#>', true, false)
			Do 'SetDDOSortedStatus'

					&TFPaletHoraFinEnsamble.FromString(DDO_Grid.FilteredText_get)
					&TFPaletHoraFinEnsamble_To.FromString(DDO_Grid.FilteredTextTo_get)
					If Not &TFPaletHoraFinEnsamble_To.IsEmpty()
						&TFPaletHoraFinEnsamble_To.Set(&TFPaletHoraFinEnsamble_To.Year(), &TFPaletHoraFinEnsamble_To.Month(), &TFPaletHoraFinEnsamble_To.Day(), 23, 59, 59)
					EndIf
				Case DDO_Grid.SelectedColumn = !'PaletNoSerie'
					&TFPaletNoSerie = DDO_Grid.FilteredText_get
					&TFPaletNoSerie_Sel = DDO_Grid.SelectedValue_get
				Case DDO_Grid.SelectedColumn = !'PaletTipo'
					&TFPaletTipo_SelsJson = DDO_Grid.SelectedValue_get
					&TFPaletTipo_Sels.FromJson(&TFPaletTipo_SelsJson)
				Case DDO_Grid.SelectedColumn = !'PaletEstatus'
					&TFPaletEstatus_SelsJson = DDO_Grid.SelectedValue_get
```

### ddo_ManageFilters.OnOptionClicked (Event)

```genexus
/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */
	Do Case
		Case ddo_ManageFilters.ActiveEventKey = !'<#Clean#>'
			Do 'CleanFilters'
			Grid.FirstPage()
			Refresh
		Case ddo_ManageFilters.ActiveEventKey = !'<#Save#>'
			Do 'SaveGridState'
			SaveFilterAs.Popup(!'Reportes.CarreteEnPalletFilters', &PgmName + !"GridState")
			&ManageFiltersExecutionStep = 2
			Refresh
		Case ddo_ManageFilters.ActiveEventKey = !'<#Manage#>'
			ManageFilters.Popup(!'Reportes.CarreteEnPalletFilters')
			&ManageFiltersExecutionStep = 2
			Refresh
		Otherwise
			&ManageFiltersXml = GetFilterByName(!'Reportes.CarreteEnPalletFilters', ddo_ManageFilters.ActiveEventKey)
			If &ManageFiltersXml.IsEmpty()
				msg('WWP_FilterNotExist')
			Else
				Do 'CleanFilters'
				SaveGridState.Call(&PgmName + !"GridState",  &ManageFiltersXml)
				&GridState.FromXml(&ManageFiltersXml)
				&OrderedBy = &GridState.OrderedBy
				&OrderedDsc = &GridState.OrderedDsc
				Do 'SetDDOSortedStatus' 
				Do 'LoadRegFiltersState'
				Grid.FirstPage()
				Refresh
			EndIf
	EndCase
	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */
```

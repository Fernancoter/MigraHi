Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'WWPBaseObjects.AuditWWColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'WWPBaseObjects.AuditWWColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each WWPBaseObjects.Audit
		order AuditDate  when &OrderedBy = 1 AND &OrderedDsc = False
		order (AuditDate)  when &OrderedBy = 1 AND &OrderedDsc = True
		order AuditId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (WWPBaseObjects.AuditId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order AuditTableName  when &OrderedBy = 3 AND &OrderedDsc = False
		order (AuditTableName)  when &OrderedBy = 3 AND &OrderedDsc = True
		order AuditTableKey  when &OrderedBy = 4 AND &OrderedDsc = False
		order (AuditTableKey)  when &OrderedBy = 4 AND &OrderedDsc = True
		order AuditDescription  when &OrderedBy = 5 AND &OrderedDsc = False
		order (AuditDescription)  when &OrderedBy = 5 AND &OrderedDsc = True
		order AuditShortDescription  when &OrderedBy = 6 AND &OrderedDsc = False
		order (AuditShortDescription)  when &OrderedBy = 6 AND &OrderedDsc = True
		order WWPBaseObjects.GAMUserGUID  when &OrderedBy = 7 AND &OrderedDsc = False
		order (Notifications.GAMUserGUID)  when &OrderedBy = 7 AND &OrderedDsc = True
		order SecUserId  when &OrderedBy = 8 AND &OrderedDsc = False
		order (Web.SecUserId)  when &OrderedBy = 8 AND &OrderedDsc = True
		order AuditAction  when &OrderedBy = 9 AND &OrderedDsc = False
		order (AuditAction)  when &OrderedBy = 9 AND &OrderedDsc = True
		
		using AuditWWDS(&FilterFullText, &TFAuditId, &TFAuditId_To, &TFAuditDate, &TFAuditDate_To, &TFAuditTableName
					, &TFAuditTableName_Sel, &TFAuditTableKey, &TFAuditTableKey_To, &TFAuditDescription, &TFAuditDescription_Sel, &TFAuditShortDescription
					, &TFAuditShortDescription_Sel, &TFGAMUserGUID, &TFGAMUserGUID_Sel, &TFSecUserId, &TFSecUserId_To, &TFAuditAction
					, &TFAuditAction_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'AuditId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = AuditId
					Case &ColumnsSelector_Column.ColumnName = !'AuditDate'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = AuditDate
					Case &ColumnsSelector_Column.ColumnName = !'AuditTableName'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(AuditTableName)
					Case &ColumnsSelector_Column.ColumnName = !'AuditTableKey'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = AuditTableKey
					Case &ColumnsSelector_Column.ColumnName = !'AuditDescription'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(Substr(AuditDescription, 1, 1000))
					Case &ColumnsSelector_Column.ColumnName = !'AuditShortDescription'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(AuditShortDescription)
					Case &ColumnsSelector_Column.ColumnName = !'GAMUserGUID'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(GAMUserGUID)
					Case &ColumnsSelector_Column.ColumnName = !'SecUserId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = SecUserId
					Case &ColumnsSelector_Column.ColumnName = !'AuditAction'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(AuditAction)
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	Else

	&OrderedDsc = &GridState.OrderedDsc

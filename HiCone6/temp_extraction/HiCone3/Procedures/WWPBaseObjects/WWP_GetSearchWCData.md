### Source (Source)

```genexus

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

If not &SearchText.IsEmpty()
	Do 'Search in transactions'
	Do 'Search Menu items'
	Do while &WWP_SearchResults.Count > &MaxCategories
		&WWP_SearchResults.Remove(&WWP_SearchResults.Count)
	Enddo
Else
	If &MaxItemsPerCategory.IsEmpty()
		Do 'Get all entities descriptions'
	EndIf
EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */


Sub 'Search in transactions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&WWP_SearchResults.Sort(!'[OrderIndex]')

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'Search Menu items'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If &AdvFilterEntities.Count = 0 OR &AdvFilterEntities.IndexOf('WWP_Search_MenuOptions') > 0
		&MenuData = MenuOptionsData()
		GetMenuAuthorizedOptions(&MenuData)

		WWP_SearchMenuOptions(&SearchText, &MenuData, &CurrentMenuOptionPath, &MenuOptions, &MenuOptionsPaths)
		If &MenuOptions.Count > 0
			&WWP_SearchResultItem = new()
			&WWP_SearchResultItem.CategoryName = 'WWP_Search_MenuOptions'
			&WWP_SearchResultItem.ShowingAllResults = True
			&WWP_SearchResults.Add(&WWP_SearchResultItem)
			&i = 1
			For &MenuDataItem in &MenuOptions
				If &WWP_SearchResultItem.Result.Count >= &MaxItemsPerCategory
					&WWP_SearchResultItem.ShowingAllResults = False
					Exit
				Else
					&SearchResult = new()
					&SearchResult.Description = &MenuDataItem.caption
					&CurrentMenuOptionPath.FromJson(&MenuOptionsPaths.Item(&i))
					If &CurrentMenuOptionPath.Count > 0
						For &OptionPath in &CurrentMenuOptionPath
							If not &SearchResult.Description2.IsEmpty()
								&SearchResult.Description2 += !' > '
							EndIf
							&SearchResult.Description2 += &OptionPath
						EndFor
						&SearchResult.Description3 = &MenuDataItem.iconClass
						&SearchResult.DisplayType = iif(&SearchResult.Description3.Trim().IsEmpty(), !'Title and subtitle', !'Title, subtitle and icon image')
					Else
						&SearchResult.Description2 = &MenuDataItem.iconClass
						&SearchResult.DisplayType = iif(&SearchResult.Description2.Trim().IsEmpty(), !'Title', !'Title and icon image')
					EndIf
					&SearchResult.Url = &MenuDataItem.link
					&WWP_SearchResultItem.Result.Add(&SearchResult)
				EndIf
				&i += 1
			EndFor
		EndIf
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'Get all entities descriptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&WWP_SearchResults.Sort(!'[OrderIndex]')
	&AdvFilterEntities = new()
	For &WWP_SearchResultItem in &WWP_SearchResults
		&AdvFilterEntities.Add(&WWP_SearchResultItem.CategoryName)
	EndFor
	&AdvFilterEntities.Add('WWP_Search_MenuOptions')

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub
```

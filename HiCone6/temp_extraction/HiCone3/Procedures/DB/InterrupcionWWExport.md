Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'DB.InterrupcionWWColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'DB.InterrupcionWWColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each Interrupcion
		order InterrupcionMotivo  when &OrderedBy = 1 AND &OrderedDsc = False
		order (InterrupcionMotivo)  when &OrderedBy = 1 AND &OrderedDsc = True
		order InterrupcionId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (InterrupcionId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order InterrupcionHoraInicio  when &OrderedBy = 3 AND &OrderedDsc = False
		order (InterrupcionHoraInicio)  when &OrderedBy = 3 AND &OrderedDsc = True
		order InterrupcionHoraFin  when &OrderedBy = 4 AND &OrderedDsc = False
		order (InterrupcionHoraFin)  when &OrderedBy = 4 AND &OrderedDsc = True
		order InterrupcionTiempo  when &OrderedBy = 5 AND &OrderedDsc = False
		order (InterrupcionTiempo)  when &OrderedBy = 5 AND &OrderedDsc = True
		order InterrupcionConcluida  when &OrderedBy = 6 AND &OrderedDsc = False
		order (InterrupcionConcluida)  when &OrderedBy = 6 AND &OrderedDsc = True
		order DownTimeCodeId  when &OrderedBy = 7 AND &OrderedDsc = False
		order (DownTimeCodeId)  when &OrderedBy = 7 AND &OrderedDsc = True
		order DownTimeCodeDescription  when &OrderedBy = 8 AND &OrderedDsc = False
		order (DB.DownTimeCodeDescription)  when &OrderedBy = 8 AND &OrderedDsc = True
		order DownTimeCodeType  when &OrderedBy = 9 AND &OrderedDsc = False
		order (DownTimeCodeType)  when &OrderedBy = 9 AND &OrderedDsc = True
		
		using InterrupcionWWDS(&FilterFullText, &TFInterrupcionId, &TFInterrupcionId_To, &TFInterrupcionHoraInicio, &TFInterrupcionHoraInicio_To, &TFInterrupcionHoraFin
					, &TFInterrupcionHoraFin_To, &TFInterrupcionTiempo, &TFInterrupcionTiempo_To, &TFInterrupcionConcluida_Sel, &TFInterrupcionMotivo, &TFInterrupcionMotivo_Sel
					, &TFDownTimeCodeId, &TFDownTimeCodeId_Sel, &TFDownTimeCodeDescription, &TFDownTimeCodeDescription_Sel, &TFDownTimeCodeType, &TFDownTimeCodeType_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = InterrupcionId
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionHoraInicio'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = InterrupcionHoraInicio
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionHoraFin'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = InterrupcionHoraFin
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionTiempo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Number = InterrupcionTiempo
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionConcluida'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = InterrupcionConcluida.ToString()
					Case &ColumnsSelector_Column.ColumnName = !'InterrupcionMotivo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(InterrupcionMotivo)
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeId)
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeDescription'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeDescription)
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeType'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeType)
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

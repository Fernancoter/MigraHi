Sub 'LoadExtrusoraNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFExtrusoraNombre = &SearchTxt
	&TFExtrusoraNombre_Sel.SetEmpty()
	For Each ExtrusoraMezcladora
		Order ExtrusoraId
		using ExtrusoraMezcladoraWWDS(&FilterFullText, &TFExtrusoraMezcladoraId, &TFExtrusoraMezcladoraId_To, &TFExtrusoraId, &TFExtrusoraId_To, &TFExtrusoraNombre
					, &TFExtrusoraNombre_Sel, &TFExtrusoraMezcladoraHusilloVirgenMin, &TFExtrusoraMezcladoraHusilloVirgenMin_To, &TFExtrusoraMezcladoraHusilloVirgenMax, &TFExtrusoraMezcladoraHusilloVirgenMax_To, &TFExtrusoraMezcladoraHusilloMolidoMin

					, &TFExtrusoraMezcladoraKgMolido_To)

		&count = 0
		For Each ExtrusoraMezcladora
			Order ExtrusoraId
			&count += 1
		EndFor
		If not ExtrusoraNombre.IsEmpty()
			&Option = ExtrusoraNombre
			
			&InsertIndex = 1
			Do while &InsertIndex <= &Options.Count AND &Options.Item(&InsertIndex) < &Option
				&InsertIndex = &InsertIndex + 1
			EndDo
			&Options.Add(&Option, &InsertIndex)
			&OptionIndexes.Add(trim(&count.ToFormattedString()), &InsertIndex)
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

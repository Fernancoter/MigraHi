Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each EmbarquePallet
		order EmbarquePalletValido  when &OrderedBy = 1
		order EmbarqueDetalleId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (EmbarqueDetalleId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order EmbarquePalletNoPallet  when &OrderedBy = 3 AND &OrderedDsc = False
		order (EmbarquePalletNoPallet)  when &OrderedBy = 3 AND &OrderedDsc = True
		order EmbarqueId  when &OrderedBy = 4 AND &OrderedDsc = False
		order (EmbarqueId)  when &OrderedBy = 4 AND &OrderedDsc = True
		
		using EmbarquePalletWWDS(&FilterFullText, &TFEmbarqueDetalleId, &TFEmbarqueDetalleId_To, &TFEmbarquePalletNoPallet, &TFEmbarquePalletNoPallet_Sel, &TFEmbarqueId

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.EmbarquePalletWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"DB.EmbarquePalletWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"DB.EmbarquePalletWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEDETALLEID"
				&TFEmbarqueDetalleId.FromString(&GridStateFilterValue.Value)
				&TFEmbarqueDetalleId_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETNOPALLET"
				&TFEmbarquePalletNoPallet.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEPALLETNOPALLET_SEL"
				&TFEmbarquePalletNoPallet_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFEMBARQUEID"
				&TFEmbarqueId.FromString(&GridStateFilterValue.Value)
				&TFEmbarqueId_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"DB.SalesPersonWWGridState") = !""

		&GridState.FromXml(&Session.Get(!"DB.SalesPersonWWGridState"))
	Endif

		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)

			Case &GridStateFilterValue.Name = !"TFSALESPERSONNAME_SEL"

	EndFor

Sub 'LoadSalesPersonNameOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFSalesPersonName = &SearchTxt
	&TFSalesPersonName_Sel.SetEmpty()
	For Each SalesPerson
		Order SalesPersonName
		using SalesPersonWWDS(&FilterFullText, &TFSalesPersonName, &TFSalesPersonName_Sel, &TFSalesPersonActive_Sel)

		Option Distinct
		If not SalesPersonName.IsEmpty()
			&Option = SalesPersonName
			
			&Options.Add(&Option)
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

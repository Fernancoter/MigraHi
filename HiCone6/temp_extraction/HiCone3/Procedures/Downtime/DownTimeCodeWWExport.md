// Exports the contents of a grid (using the selected order and filters) to an Excel file.

Sub 'OpenDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&Random = Random() * 10000
	&Filename = !"DownTimeCodeWWExport-" + &Random.ToString().Trim() + !".xlsx"

	&ExcelDocument.Open(&Filename)
	Do 'CheckStatus'
	&ExcelDocument.Clear()

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())

		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&FilterFullText)
	EndIf
	If not (&TFDownTimeCodeId_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Id")

	Else
		If not (&TFDownTimeCodeId.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Id")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFDownTimeCodeId)
		EndIf
	EndIf
	If not (&TFDownTimeCodeName_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Name")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFDownTimeCodeName_Sel)
	Else
		If not (&TFDownTimeCodeName.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Name")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFDownTimeCodeName)
		EndIf
	EndIf
	If not (&TFDownTimeCodeDescription_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Description")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFDownTimeCodeDescription_Sel)
	Else
		If not (&TFDownTimeCodeDescription.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Code Description")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFDownTimeCodeDescription)
		EndIf
	EndIf
	&CellRow += 2

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'Downtime.DownTimeCodeWWColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'Downtime.DownTimeCodeWWColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DownTimeCode
		order DownTimeCodeName  when &OrderedBy = 1 AND &OrderedDsc = False
		order (DownTimeCodeName)  when &OrderedBy = 1 AND &OrderedDsc = True
		order DownTimeCodeId  when &OrderedBy = 2 AND &OrderedDsc = False
		order (DownTimeCodeId)  when &OrderedBy = 2 AND &OrderedDsc = True
		order DownTimeCodeDescription  when &OrderedBy = 3 AND &OrderedDsc = False
		order (DownTimeCodeDescription)  when &OrderedBy = 3 AND &OrderedDsc = True
		
		using DownTimeCodeWWDS(&FilterFullText, &TFDownTimeCodeId, &TFDownTimeCodeId_Sel, &TFDownTimeCodeName, &TFDownTimeCodeName_Sel, &TFDownTimeCodeDescription
					, &TFDownTimeCodeDescription_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeId'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeId)
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeName'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeName)
					Case &ColumnsSelector_Column.ColumnName = !'DownTimeCodeDescription'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(DownTimeCodeDescription)
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"DownTimeCodeId", '', !'Code Id', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"DownTimeCodeName", '', !'Code Name', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"DownTimeCodeDescription", '', !'Code Description', True, '')
		
	&UserCustomValue = LoadColumnsSelectorState.Udp(!'Downtime.DownTimeCodeWWColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Downtime.DownTimeCodeWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Downtime.DownTimeCodeWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Downtime.DownTimeCodeWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEID"
				&TFDownTimeCodeId.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEID_SEL"
				&TFDownTimeCodeId_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODENAME"
				&TFDownTimeCodeName.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODENAME_SEL"
				&TFDownTimeCodeName_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEDESCRIPTION"
				&TFDownTimeCodeDescription.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFDOWNTIMECODEDESCRIPTION_SEL"
				&TFDownTimeCodeDescription_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

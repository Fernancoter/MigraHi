Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Calidad.reclamoswwGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Calidad.reclamoswwGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Calidad.reclamoswwGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMOFECHA"
				&TFReclamoFecha.FromString(&GridStateFilterValue.Value)
				&TFReclamoFecha_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFRECLAMOCODIGO"
				&TFReclamoCodigo.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMOCODIGO_SEL"
				&TFReclamoCodigo_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMOCLIENTE"
				&TFReclamoCliente.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMOCLIENTE_SEL"
				&TFReclamoCliente_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPLANTACLIENTENOMBRE"
				&TFPlantaClienteNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFPLANTACLIENTENOMBRE_SEL"
				&TFPlantaClienteNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODESCRIPCION"
				&TFReclamoDescripcion.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODESCRIPCION_SEL"
				&TFReclamoDescripcion_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMOCARRETESREPORTADOS"
				&TFReclamoCarretesReportados.FromString(&GridStateFilterValue.Value)
				&TFReclamoCarretesReportados_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFRECLAMOMILLARESREPORTADOS"
				&TFReclamoMillaresReportados.FromString(&GridStateFilterValue.Value)
				&TFReclamoMillaresReportados_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFRECLAMOFECHACIERRE"
				&TFReclamoFechaCierre.FromString(&GridStateFilterValue.Value)
				&TFReclamoFechaCierre_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFRECLAMONOTACREDITO"

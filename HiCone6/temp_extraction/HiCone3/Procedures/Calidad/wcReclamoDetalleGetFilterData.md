Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Calidad.wcReclamoDetalleGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Calidad.wcReclamoDetalleGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Calidad.wcReclamoDetalleGridState"))
	Endif	

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLECODIGO"
				&TFReclamoDetalleCodigo.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLECODIGO_SEL"
				&TFReclamoDetalleCodigo_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEPRODUCTONOMBRE"
				&TFReclamoDetalleProductoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEPRODUCTONOMBRE_SEL"
				&TFReclamoDetalleProductoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEESTADO_SEL"
				&TFReclamoDetalleEstado_SelsJson = &GridStateFilterValue.Value
				&TFReclamoDetalleEstado_Sels.FromJson(&TFReclamoDetalleEstado_SelsJson)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEMILLAR"
				&TFReclamoDetalleMillar.FromString(&GridStateFilterValue.Value)
				&TFReclamoDetalleMillar_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEOBSERVACIONRCA"
				&TFReclamoDetalleObservacionRCA.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFRECLAMODETALLEOBSERVACIONRCA_SEL"

		EndCase
	EndFor

Sub 'LoadReclamoDetalleCodigoOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFReclamoDetalleCodigo = &SearchTxt
	&TFReclamoDetalleCodigo_Sel.SetEmpty()
	For Each ReclamoDetalle
		Order ReclamoDetalleCodigo
		using wcReclamoDetalleDS(&TFReclamoDetalleCodigo, &TFReclamoDetalleCodigo_Sel, &TFReclamoDetalleProductoNombre, &TFReclamoDetalleProductoNombre_Sel, &TFReclamoDetalleEstado_Sels, &TFReclamoDetalleMillar
					, &TFReclamoDetalleMillar_To, &TFReclamoDetalleObservacionRCA, &TFReclamoDetalleObservacionRCA_Sel)
		Where ReclamoId = &ReclamoId

		&count = 0
		For Each ReclamoDetalle
			Order ReclamoDetalleCodigo
			&count += 1
		EndFor
		If not ReclamoDetalleCodigo.IsEmpty()
			&Option = ReclamoDetalleCodigo
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

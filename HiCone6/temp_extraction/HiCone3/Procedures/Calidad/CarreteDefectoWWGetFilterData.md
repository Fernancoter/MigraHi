Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Calidad.CarreteDefectoWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Calidad.CarreteDefectoWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Calidad.CarreteDefectoWWGridState"))
	Endif	

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCARRETEDEFECTOID"
				&TFCarreteDefectoId.FromString(&GridStateFilterValue.Value)
				&TFCarreteDefectoId_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFCARRETEDEFECTONOMBRE"
				&TFCarreteDefectoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCARRETEDEFECTONOMBRE_SEL"
				&TFCarreteDefectoNombre_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadCarreteDefectoNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFCarreteDefectoNombre = &SearchTxt
	&TFCarreteDefectoNombre_Sel.SetEmpty()
	For Each CarreteDefecto
		Order CarreteDefectoNombre
		using CarreteDefectoWWDS(&FilterFullText, &TFCarreteDefectoId, &TFCarreteDefectoId_To, &TFCarreteDefectoNombre, &TFCarreteDefectoNombre_Sel)

		&count = 0
		For Each CarreteDefecto
			Order CarreteDefectoNombre
			&count += 1
		EndFor
		If not CarreteDefectoNombre.IsEmpty()
			&Option = CarreteDefectoNombre
			
			&Options.Add(&Option)
			&OptionIndexes.Add(trim(&count.ToFormattedString()))
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

// Exports the contents of a grid (using the selected order and filters) to an Excel file.
LoadWWPContext.Call(&WWPContext)

Do 'OpenDocument'

&CellRow = 1
&FirstColumn = 1

Do 'LoadGridState'

Do 'WriteFilters'

Sub 'OpenDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&Random = Random() * 10000
	&Filename = !"CausaInterrupcionWWExport-" + &Random.ToString().Trim() + !".xlsx"

	&ExcelDocument.Open(&Filename)
	Do 'CheckStatus'
	&ExcelDocument.Clear()

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "WWP_FullTextFilterDescription")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&FilterFullText)
	EndIf
	If not (&TFCausaInterrupcionNombre_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Nombre")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFCausaInterrupcionNombre_Sel)
	Else
		If not (&TFCausaInterrupcionNombre.IsEmpty())
			WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Nombre")
			&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&TFCausaInterrupcionNombre)
		EndIf
	EndIf
	If not (&TFCausaInterrupcionPrensa_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Prensa")
		Do Case
			Case &TFCausaInterrupcionPrensa_Sel = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSChecked"
			Case &TFCausaInterrupcionPrensa_Sel = 2
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSUnChecked"
		EndCase
	EndIf
	If not (&TFCausaInterrupcionExtrusora_Sel.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Extrusora")
		Do Case
			Case &TFCausaInterrupcionExtrusora_Sel = 1
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSChecked"
			Case &TFCausaInterrupcionExtrusora_Sel = 2
				&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = "WWP_TSUnChecked"
		EndCase
	EndIf
	&CellRow += 2

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'Reportes.CausaInterrupcionWWColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'Reportes.CausaInterrupcionWWColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each CausaInterrupcion
		order CausaInterrupcionNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (CausaInterrupcionNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order CausaInterrupcionPrensa  when &OrderedBy = 2 AND &OrderedDsc = False
		order (CausaInterrupcionPrensa)  when &OrderedBy = 2 AND &OrderedDsc = True
		order CausaInterrupcionExtrusora  when &OrderedBy = 3 AND &OrderedDsc = False
		order (CausaInterrupcionExtrusora)  when &OrderedBy = 3 AND &OrderedDsc = True
		
		using Reportes.CausaInterrupcionWWDS(&FilterFullText, &TFCausaInterrupcionNombre, &TFCausaInterrupcionNombre_Sel, &TFCausaInterrupcionPrensa_Sel, &TFCausaInterrupcionExtrusora_Sel)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'CausaInterrupcionNombre'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = WWP_Export_SecureText(CausaInterrupcionNombre)
					Case &ColumnsSelector_Column.ColumnName = !'CausaInterrupcionPrensa'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = CausaInterrupcionPrensa.ToString()
					Case &ColumnsSelector_Column.ColumnName = !'CausaInterrupcionExtrusora'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = CausaInterrupcionExtrusora.ToString()
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'CloseDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ExcelDocument.Save()
	Do 'CheckStatus'
	&ExcelDocument.Close()

	If (&ExcelDocument.ErrCode <> 0)
		&Filename = ""
		&ErrorMessage = &ExcelDocument.ErrDescription
		&ExcelDocument.Close()
		Return
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"CausaInterrupcionNombre", '', !'Nombre', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"CausaInterrupcionPrensa", '', !'Prensa', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"CausaInterrupcionExtrusora", '', !'Extrusora', True, '')
		
	&UserCustomValue = WWPBaseObjects.LoadColumnsSelectorState.Udp(!'Reportes.CausaInterrupcionWWColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Reportes.CausaInterrupcionWWGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Reportes.CausaInterrupcionWWGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Reportes.CausaInterrupcionWWGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCAUSAINTERRUPCIONNOMBRE"
				&TFCausaInterrupcionNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCAUSAINTERRUPCIONNOMBRE_SEL"
				&TFCausaInterrupcionNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCAUSAINTERRUPCIONPRENSA_SEL"
				&TFCausaInterrupcionPrensa_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFCAUSAINTERRUPCIONEXTRUSORA_SEL"
				&TFCausaInterrupcionExtrusora_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

EndSub

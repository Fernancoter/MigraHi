	For each DB.Product
		order ProductDesc  when &OrderedBy = 1 AND &OrderedDsc = False
		order (ProductDesc)  when &OrderedBy = 1 AND &OrderedDsc = True
		order ProductNumber  when &OrderedBy = 2 AND &OrderedDsc = False

		order Product8020  when &OrderedBy = 11 AND &OrderedDsc = False

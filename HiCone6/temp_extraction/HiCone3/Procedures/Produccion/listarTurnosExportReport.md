	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not &FilterFullText.IsEmpty()
		print printBlockFilterFilterFullText
	EndIf
	If not &TFTurnoNombre_Sel.IsEmpty()

	Else

Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Turno
		order TurnoId  when &OrderedBy = 1
		order TurnoNombre  when &OrderedBy = 2 AND &OrderedDsc = False
		order (TurnoNombre)  when &OrderedBy = 2 AND &OrderedDsc = True
		order TurnoHoraInicio  when &OrderedBy = 3 AND &OrderedDsc = False
		order (TurnoHoraInicio)  when &OrderedBy = 3 AND &OrderedDsc = True
		order TurnoHoraFin  when &OrderedBy = 4 AND &OrderedDsc = False
		order (TurnoHoraFin)  when &OrderedBy = 4 AND &OrderedDsc = True
		
		using listarTurnosDS(&FilterFullText, &TFTurnoNombre, &TFTurnoNombre_Sel, &TFTurnoHoraInicio, &TFTurnoHoraInicio_To, &TFTurnoHoraFin
					, &TFTurnoHoraFin_To)

		Do 'BeforePrintLine'
		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarTurnosGridState") = !""
		&GridState.FromXml(WWPBaseObjects.LoadGridState.Udp(!"Produccion.listarTurnosGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarTurnosGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE"
				&TFTurnoNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNONOMBRE_SEL"
				&TFTurnoNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFTURNOHORAINICIO"
				&TFTurnoHoraInicio.FromString(&GridStateFilterValue.Value)
				&TFTurnoHoraInicio_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFTURNOHORAFIN"
				&TFTurnoHoraFin.FromString(&GridStateFilterValue.Value)
				&TFTurnoHoraFin_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

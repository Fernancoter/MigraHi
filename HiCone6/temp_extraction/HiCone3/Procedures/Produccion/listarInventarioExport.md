	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */


Sub 'OpenDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&Random = Random() * 10000
	&Filename = !"listarInventarioExport-" + &Random.ToString().Trim() + !".xlsx"

	&ExcelDocument.Open(&Filename)
	Do 'CheckStatus'
	&ExcelDocument.Clear()

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not (&FilterFullText.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "WWP_FullTextFilterDescription")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Text = WWP_Export_SecureText(&FilterFullText)
	EndIf
	If not (&TFInventarioFechaHora.IsEmpty() AND &TFInventarioFechaHora_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Fecha Hora de Reinicio de Consecutivo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Date = &TFInventarioFechaHora
		WWPBaseObjects.WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Date = &TFInventarioFechaHora_To
	EndIf
	If not (&TFInventarioInicioConsecutivo.IsEmpty() AND &TFInventarioInicioConsecutivo_To.IsEmpty())
		WWP_ExportWriteFilter(&ExcelDocument, true, &CellRow, &FirstColumn, "Inicio Consecutivo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 1).Date = &TFInventarioInicioConsecutivo
		WWP_ExportWriteFilter(&ExcelDocument, false, &CellRow, &FirstColumn + 2, "WWP_TSTo")
		&ExcelDocument.Cells(&CellRow, &FirstColumn + 3).Date = &TFInventarioInicioConsecutivo_To
	EndIf
	&CellRow += 2

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteColumnTitles'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&VisibleColumnCount = 0
	If &Session.Get(!'Produccion.listarInventarioColumnsSelector') <> ''
		&ColumnsSelectorXML = &Session.Get(!'Produccion.listarInventarioColumnsSelector')
		&ColumnsSelector.FromXml(&ColumnsSelectorXML)
	Else
		Do 'InitializeColumnsSelector'
	EndIf

	&ColumnsSelector.Columns.Sort(!'Order')
	For &ColumnsSelector_Column in &ColumnsSelector.Columns
		If &ColumnsSelector_Column.IsVisible = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Text = GetMessageText(iif(&ColumnsSelector_Column.DisplayName.IsEmpty(), &ColumnsSelector_Column.ColumnName, &ColumnsSelector_Column.DisplayName))
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Bold = True
			&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Color = 11
			&VisibleColumnCount += 1
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'WriteData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Inventario
		order InventarioFechaHora  when &OrderedBy = 1 AND &OrderedDsc = False
		order (InventarioFechaHora)  when &OrderedBy = 1 AND &OrderedDsc = True
		order InventarioInicioConsecutivo  when &OrderedBy = 2 AND &OrderedDsc = False
		order (InventarioInicioConsecutivo)  when &OrderedBy = 2 AND &OrderedDsc = True
		
		using listarInventarioDS(&FilterFullText, &TFInventarioFechaHora, &TFInventarioFechaHora_To, &TFInventarioInicioConsecutivo, &TFInventarioInicioConsecutivo_To)

		// Write cell values
		&CellRow += 1
		
		Do 'BeforeWriteLine'
		&VisibleColumnCount = 0
		For &ColumnsSelector_Column in &ColumnsSelector.Columns
			If &ColumnsSelector_Column.IsVisible = True
				Do Case
					Case &ColumnsSelector_Column.ColumnName = !'InventarioFechaHora'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = InventarioFechaHora
					Case &ColumnsSelector_Column.ColumnName = !'InventarioInicioConsecutivo'
						&ExcelDocument.Cells(&CellRow, &FirstColumn + &VisibleColumnCount).Date = InventarioInicioConsecutivo
				EndCase
				&VisibleColumnCount += 1
			EndIf
		EndFor		
		
		Do 'AfterWriteLine'

	Endfor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'CloseDocument'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ExcelDocument.Save()
	Do 'CheckStatus'
	&ExcelDocument.Close()

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'CheckStatus'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If (&ExcelDocument.ErrCode <> 0)
		&Filename = ""
		&ErrorMessage = &ExcelDocument.ErrDescription
		&ExcelDocument.Close()
		Return
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'InitializeColumnsSelector'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&ColumnsSelector = new WWPColumnsSelector()
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"InventarioFechaHora", '', !'Fecha Hora de Reinicio de Consecutivo', True, '')
	WWP_ColumnsSelector_Add(&ColumnsSelector, !"InventarioInicioConsecutivo", '', !'Inicio Consecutivo', True, '')
		
	&UserCustomValue = LoadColumnsSelectorState.Udp(!'Produccion.listarInventarioColumnsSelector')
	If not(&UserCustomValue.IsEmpty())
		&ColumnsSelectorAux.FromXml(&UserCustomValue)
		WWPBaseObjects.WWP_ColumnSelector_UpdateColumns(&ColumnsSelectorAux, &ColumnsSelector)
	EndIf

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarInventarioGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarInventarioGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarInventarioGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFINVENTARIOFECHAHORA"
				&TFInventarioFechaHora.FromString(&GridStateFilterValue.Value)
				&TFInventarioFechaHora_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFINVENTARIOINICIOCONSECUTIVO"
				&TFInventarioInicioConsecutivo.FromString(&GridStateFilterValue.Value)
				&TFInventarioInicioConsecutivo_To.FromString(&GridStateFilterValue.ValueTo)
		EndCase
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */



	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

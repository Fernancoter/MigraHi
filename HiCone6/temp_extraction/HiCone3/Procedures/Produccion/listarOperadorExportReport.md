Sub 'PrintFilters'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	If not &FilterFullText.IsEmpty()
		print printBlockFilterFilterFullText
	EndIf
	If not &TFOperadorNombre_Sel.IsEmpty()
		print printBlockTFOperadorNombre_Sel
	Else
		If not &TFOperadorNombre.IsEmpty()
			print printBlockTFOperadorNombre
		EndIf
	EndIf
	If not &TFOperadorActivo_Sel.IsEmpty()
		Do Case
			Case &TFOperadorActivo_Sel = 1
				&FilterTFOperadorActivo_SelValueDescription = "WWP_TSChecked" 
			Case &TFOperadorActivo_Sel = 2
				&FilterTFOperadorActivo_SelValueDescription = "WWP_TSUnChecked"

	EndIf

Sub 'PrintData'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Operador
		order OperadorNombre  when &OrderedBy = 1 AND &OrderedDsc = False
		order (OperadorNombre)  when &OrderedBy = 1 AND &OrderedDsc = True
		order OperadorActivo  when &OrderedBy = 2 AND &OrderedDsc = False
		order (OperadorActivo)  when &OrderedBy = 2 AND &OrderedDsc = True
		
		using listarOperadorDS(&FilterFullText, &TFOperadorNombre, &TFOperadorNombre_Sel, &TFOperadorActivo_Sel)

		Do 'BeforePrintLine'
		print printBlockLines_data
		Do 'AfterPrintLine'
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarOperadorGridState") = !""
		&GridState.FromXml(WWPBaseObjects.LoadGridState.Udp(!"Produccion.listarOperadorGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarOperadorGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFOPERADORNOMBRE"
				&TFOperadorNombre.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFOPERADORNOMBRE_SEL"
				&TFOperadorNombre_Sel.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFOPERADORACTIVO_SEL"
				&TFOperadorActivo_Sel.FromString(&GridStateFilterValue.Value)
		EndCase
	EndFor

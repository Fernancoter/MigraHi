	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	For each DB.Inventario
		order DB.InventarioFechaHora  when &OrderedBy = 1 AND &OrderedDsc = False

		order InventarioInicioConsecutivo  when &OrderedBy = 2 AND &OrderedDsc = False
		order (InventarioInicioConsecutivo)  when &OrderedBy = 2 AND &OrderedDsc = True

Sub 'LoadGridState'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	// Load grid state from session.
	If &Session.Get(!"Produccion.listarInventarioGridState") = !""
		&GridState.FromXml(LoadGridState.Udp(!"Produccion.listarInventarioGridState"))
	Else
		&GridState.FromXml(&Session.Get(!"Produccion.listarInventarioGridState"))
	Endif	

	&OrderedBy = &GridState.OrderedBy
	&OrderedDsc = &GridState.OrderedDsc

	For &GridStateFilterValue in &GridState.FilterValues
		Do Case
			Case &GridStateFilterValue.Name = !"FILTERFULLTEXT"
				&FilterFullText.FromString(&GridStateFilterValue.Value)
			Case &GridStateFilterValue.Name = !"TFINVENTARIOFECHAHORA"
				&TFInventarioFechaHora.FromString(&GridStateFilterValue.Value)
				&TFInventarioFechaHora_To.FromString(&GridStateFilterValue.ValueTo)
			Case &GridStateFilterValue.Name = !"TFINVENTARIOINICIOCONSECUTIVO"
				&TFInventarioInicioConsecutivo.FromString(&GridStateFilterValue.Value)
				&TFInventarioInicioConsecutivo_To.FromString(&GridStateFilterValue.ValueTo)

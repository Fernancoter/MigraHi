Sub 'LoadExtrusionExtrusoraNombreOptions'

	/* Generated by DVelop Work With Plus Pattern [Start] - Do not change */

	&TFExtrusionExtrusoraNombre = &SearchTxt
	&TFExtrusionExtrusoraNombre_Sel.SetEmpty()
	For Each DB.Extrusion
		Order ExtrusionExtrusoraId
		using TableroDirectivoExtrusionDS(&TFExtrusionFecha, &TFExtrusionFecha_To, &TFExtrusionExtrusoraNombre, &TFExtrusionExtrusoraNombre_Sel, &TFExtrusionTurnoNombre, &TFExtrusionTurnoNombre_Sel
					, &TFExtrusionProductoNombre, &TFExtrusionProductoNombre_Sel, &TFExtrusionOperadorNombre, &TFExtrusionOperadorNombre_Sel, &TFExtrusionMeta, &TFExtrusionMeta_To
					, &TFExtrusionBobinas, &TFExtrusionBobinas_To)
		Where DB.ExtrusionEstado in (EstadoExtrusion.Programada, EstadoExtrusion.Terminada)

		&count = 0
		For Each DB.Extrusion
			Order ExtrusionExtrusoraId
			&count += 1
		EndFor
		If not ExtrusionExtrusoraNombre.IsEmpty()
			&Option = ExtrusionExtrusoraNombre
			
			&InsertIndex = 1
			Do while &InsertIndex <= &Options.Count AND &Options.Item(&InsertIndex) < &Option
				&InsertIndex = &InsertIndex + 1
			EndDo
			&Options.Add(&Option, &InsertIndex)
			&OptionIndexes.Add(trim(&count.ToFormattedString()), &InsertIndex)
		EndIf
		If &Options.Count = 50
			Exit
		EndIf
	EndFor

	/* Generated by DVelop Work With Plus Pattern [End] - Do not change */

EndSub
